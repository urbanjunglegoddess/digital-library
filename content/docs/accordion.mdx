      type AccordionMode = "single" | "multiple";

      interface AccordionCtx {
        openIds: Set<string>;
        toggle: (id: string) => void;
        mode: AccordionMode;
      }
      const Ctx = createContext<AccordionCtx | null>(null);
      const useAccordion = () => {
        const c = useContext(Ctx);
        if (!c) throw new Error("Accordion.* must be used inside <Accordion>");
        return c;
      };

      interface RootProps {
        mode?: AccordionMode;
        defaultOpen?: string[];
        collapsible?: boolean;          // single-mode: allow closing the open one
        children: ReactNode;
      }

      export function Accordion({ mode = "single", defaultOpen = [], collapsible = true, children }: RootProps) {
        const [openIds, setOpenIds] = useState<Set<string>>(new Set(defaultOpen));

        const toggle = useCallback((id: string) => {
          setOpenIds((prev) => {
            const next = new Set(prev);
            const isOpen = next.has(id);
            if (mode === "single") {
              next.clear();
              if (!isOpen) next.add(id);
              else if (!collapsible) next.add(id);   // not collapsible => keep it open
            } else {
              isOpen ? next.delete(id) : next.add(id);
            }
            return next;
          });
        }, [mode, collapsible]);

        return <Ctx.Provider value={{ openIds, toggle, mode }}><div className="accordion">{children}</div></Ctx.Provider>;
      }

      interface ItemCtx { headerId: string; panelId: string; id: string; open: boolean; }
      const ItemContext = createContext<ItemCtx | null>(null);
      const useItem = () => {
        const c = useContext(ItemContext);
        if (!c) throw new Error("Accordion.Header/Panel must be inside Accordion.Item");
        return c;
      };

      export function AccordionItem({ id, children }: { id?: string; children: ReactNode }) {
        const auto = useId();
        const itemId = id ?? auto;
        const { openIds } = useAccordion();
        const value: ItemCtx = {
          id: itemId,
          headerId: `${itemId}-header`,
          panelId: `${itemId}-panel`,
          open: openIds.has(itemId),
        };
        return <ItemContext.Provider value={value}>{children}</ItemContext.Provider>;
      }

      export function AccordionHeader({ children, level = 3 }: { children: ReactNode; level?: 2 | 3 | 4 }) {
        const { toggle } = useAccordion();
        const { id, headerId, panelId, open } = useItem();
        const Heading = `h${level}` as const;

        // Arrow Up/Down + Home/End rove focus BETWEEN headers (APG-recommended). Enter/Space
        // fall through to native <button> activation — we never preventDefault them.
        const onKey = (e: ReactKeyboardEvent<HTMLButtonElement>) => {
          const root = e.currentTarget.closest(".accordion");
          if (!root) return;
          const triggers = [...root.querySelectorAll<HTMLButtonElement>(".accordion__trigger")];
          const i = triggers.indexOf(e.currentTarget);
          let next: HTMLButtonElement | undefined;
          switch (e.key) {
            case "ArrowDown": next = triggers[(i + 1) % triggers.length]; break;
            case "ArrowUp":   next = triggers[(i - 1 + triggers.length) % triggers.length]; break;
            case "Home":      next = triggers[0]; break;
            case "End":       next = triggers[triggers.length - 1]; break;
            default: return;                        // Enter/Space toggle via the native button
          }
          e.preventDefault();
          next?.focus();
        };

        return (
          <Heading className="accordion__header">
            <button
              type="button"
              id={headerId}
              className="accordion__trigger"
              aria-expanded={open}
              aria-controls={panelId}
              onClick={() => toggle(id)}
              onKeyDown={onKey}
            >
              <span>{children}</span>
              <svg className="chevron" aria-hidden viewBox="0 0 20 20"><path d="M6 8l4 4 4-4" /></svg>
            </button>
          </Heading>
        );
      }

      export function AccordionPanel({ children }: { children: ReactNode }) {
        const { headerId, panelId, open } = useItem();
        return (
          <div
            id={panelId}
            role="region"
            aria-labelledby={headerId}
            className="accordion__panel"
            hidden={!open}          // hidden pulls focusables out of tab order
          >
            <div className="accordion__inner">{children}</div>
          </div>
        );
      }

## 1\. What an Accordion Actually Is
An accordion is a **vertical stack of headers, each of which expands to reveal a panel of content and collapses to hide it again.** That's it. But the word gets slapped onto three genuinely different components, and picking the wrong one is the root of most "my accordion is broken" bugs:

*   **Accordion (the real thing):** a _set_ of stacked, independently collapsible sections. Multiple headers, each owning a panel. Section 2 knows about section 1 and section 3 — they're siblings in one control, and often only one is open at a time.
*   **Disclosure (a.k.a. a single** **`<details>`****\*\*\*\*):** _one_ header, _one_ panel. A lone show/hide toggle. "Read more," a single FAQ item, an optional-settings reveal. An accordion is, structurally, a group of disclosures with shared behavior. If you only have one collapsible section, you don't have an accordion — you have a disclosure, and you should stop reaching for an accordion library.
*   **Tabs:** also "one panel visible at a time," but tabs sit _side by side_ in a horizontal strip and the panels occupy the _same_ region below. Tabs never let you see two panels at once and never collapse everything to nothing. An accordion stacks vertically, can show many panels (multi-open), and can show _zero_ panels (all collapsed).

The distinction that trips people up most: **accordion vs tabs.** They feel interchangeable and they are not. Rule of thumb — if the content is peer sections a user scans and dips into (an FAQ, a spec sheet, a settings group), use an accordion. If the content is mutually-exclusive views of the _same_ subject (Overview / Reviews / Shipping on a product), use tabs. Accordions are for _browsing a list_; tabs are for _switching a view_. On narrow/mobile screens, tabs frequently collapse _into_ accordions — which is exactly why understanding both matters.

And the distinction that trips up _engineers_: **native** **`<details>`** **vs the ARIA-button pattern.** These are the two legitimate ways to build the thing. `<details>`/`<summary>` is a real browser primitive with open/close behavior baked in — zero JS to toggle. The ARIA pattern is a heading + a `<button aria-expanded>` + a panel you wire up yourself. Both are correct. Choosing between them is §8 and §9's central decision, and getting it wrong is where accessibility quietly dies.

## 2\. Why It Matters
The accordion is a **density negotiator** — it trades immediate visibility for a shorter page. That trade has real stakes:

*   **It shapes what users find.** Content inside a collapsed panel is, to a first approximation, _not seen_ and _not searched_ by the human eye (Ctrl-F still finds it in native/`hidden="until-found"` implementations — more on that later). Bury your pricing table or your return policy in a collapsed accordion and you've hidden the exact thing people came for. Every collapse is a bet that the header is a good-enough summary.
*   **Accessibility is unusually easy to get wrong here.** An accordion is a heading, a button, a controlled region, a keyboard contract, and focus management _all at once_. The single most common failure in the wild is a `<div onClick>` header with a rotating chevron and no `aria-expanded`, no heading semantics, and no keyboard support. Screen-reader users hit a wall. Get the accordion right and you've demonstrated command of half the ARIA spec.
*   **It's an SEO and Core Web Vitals surface.** Google indexes content inside collapsed accordions, but layout that shifts when panels animate open tanks your **CLS** (Cumulative Layout Shift) score if you're careless. And the height-animation problem (§9) is where junior builds either don't animate at all or animate in a way that thrashes layout every frame.
*   **It sets the "is this app polished?" tone.** A snappy, correctly-animated accordion with a rotating icon and smooth height transition reads as _craft_. A jumpy, instant-snap, or janky-easing accordion reads as unfinished — even when the content is perfect. Perceived quality lives in these micro-transitions.

## 3\. Anatomy of an Accordion
Every well-built accordion decomposes into a predictable set of parts. Name them precisely — the ARIA relationships depend on it:

*   **Root / container:** the wrapping element that owns the group. Holds the shared state (which items are open) and the mode (single vs multi). Often `role="presentation"` or just a plain `<div>`; the semantics live on the items.
*   **Item:** one collapsible unit = one header + one panel, bound together.
*   **Header (the heading wrapper):** a real heading element (`<h2>`/`<h3>`) whose _only_ child is the trigger button. This is the part juniors skip. The heading level communicates document structure to screen-reader users navigating by heading; the button inside it does the toggling. Heading and button are two jobs, two elements.
*   **Trigger (the button):** the interactive control. Carries `aria-expanded` (true/false) and `aria-controls` (pointing at the panel's id). In native builds this is the `<summary>`.
*   **Icon / indicator:** the chevron, plus/minus, or caret that rotates or swaps to signal state. Decorative → `aria-hidden="true"`. State is announced by `aria-expanded`, never by the icon.
*   **Panel / region:** the collapsible content. Carries `role="region"` and `aria-labelledby` pointing back at its trigger, so a screen reader announces "Shipping, region" when focus lands inside. Hidden with `hidden` (or `display:none` / `content-visibility`) when collapsed — not merely visually clipped, or its focusable children stay tabbable while invisible.
*   **The animated wrapper (implementation detail):** because you can't animate to `height: auto` the naive way, real builds insert a wrapper (the grid-row track, or the `::details-content` pseudo) whose sole job is to make the open/close transition smooth. It carries no semantics.

The relationship that must be airtight: **button** **`aria-controls`** **→ panel** **`id`**, and **panel** **`aria-labelledby`** **→ button** **`id`**. Break either and screen-reader users lose the thread.

## 4\. Sizes, Density & Scale
Accordions don't come in "sizes" the way buttons do, but density is a real design axis. The header is the primary tap target and the number one thing to get right dimensionally.

| Density | Header height | Header padding (V) | Header font | Panel padding | Use |
| ---| ---| ---| ---| ---| --- |
| Compact | 40px | 8px | 14px | 12px 16px | Dense settings, admin panels, nested trees |
| Default | 48px | 12px | 15–16px | 16px | Most UI, FAQs, filter groups |
| Comfortable | 56px | 16px | 16–18px | 20px 24px | Marketing FAQs, mobile-first, touch-heavy |
| Large / editorial | 64px+ | 20px+ | 18–20px | 24px 32px | Landing pages, hero FAQs, high-emphasis |

Rules that matter:

*   **The header's tap target stays ≥44px** no matter how "compact" the visual. Shrink the font, keep the hit area — expand it with padding, not a smaller click zone.
*   **Header height must not jump when the icon swaps** (plus→minus glyphs often have different metrics). Fix the icon box dimensions so the header doesn't reflow on toggle.
*   **Drive spacing from tokens**, not magic numbers, so compact/comfortable are a token swap, not a rewrite.
*   **Panel padding lives on an inner wrapper, not the animated element.** If you put vertical padding directly on the collapsing element, the grid/height animation fights the padding and you get a jump at the end. Padding goes on a child; the animated wrapper is padding-free (this is the single most common animation bug — §9.4).

## 5\. States (the part juniors forget)
An accordion multiplies button states by open/closed, and adds a couple of its own. Missing states are the #1 tell of an amateur build.

**Per header (trigger):**

1. **Collapsed (rest)** — panel hidden, `aria-expanded="false"`, icon in its "closed" orientation.
2. **Expanded** — panel visible, `aria-expanded="true"`, icon rotated/swapped.
3. **Hover** — subtle background shift on the header. Never the _only_ signal of interactivity.
4. **Focus** — reachable by keyboard, visible `:focus-visible` ring on the trigger. Non-negotiable.
5. **Active (pressed)** — the moment of toggle; a slight background darken sells the click.
6. **Disabled** — this section can't be opened right now (e.g. a checkout step not yet reachable). `aria-disabled="true"`, reduced emphasis, and — critically — _explain why nearby_, because a dead accordion header is baffling otherwise.

**Per panel / whole component:**

1. **Opening / closing (in transition)** — the animated middle. Content must not be interactable mid-collapse; focus must not be trapped in a half-closed panel.
2. **Loading** — panel content fetched on expand (lazy accordion). Show a skeleton or spinner _inside_ the panel while the header stays responsive.
3. **Empty** — an expanded panel with nothing in it. Say so ("No items"), don't render a silent void.
4. **Error** — lazy-loaded content failed. Show a retry affordance inside the panel; keep the header openable.

**Component-level:**

1. **All collapsed** — a legitimate resting state for a multi-open accordion (and impossible for tabs). Design for it; don't assume something is always open.
2. **Single-open constraint active** — opening one closes the last (§6). The _closing_ of the sibling is a state transition users see; animate it or it feels glitchy.

## 6\. Types & Variants
Accordions vary along two independent axes: **how many can be open** and **how it's built/styled**.

**By open-behavior (the important one):**

*   **Single-open (exclusive):** opening any section closes the others. At most one panel open at a time. Good when panels are long and you want a bounded page height. This is the classic "accordion" behavior and what the native `name` attribute now gives you for free (§9.1). Downside: users can't compare two sections side by side.
*   **Multi-open (independent):** every section toggles on its own; any number open at once. Good for FAQs and checklists where people want to open several and keep them open. Downside: the page can grow tall and unbounded.
*   **Single-open with "allow all closed":** exclusive, but clicking the open header closes it, leaving nothing open. Almost always the right flavor of single-open — never trap the user with "something must always be open."

**By structure/emphasis:**

*   **Bordered / boxed:** each item in its own card with a border. Reads as discrete units. The FAQ default.
*   **Flush / divided:** items separated only by hairline rules, no outer border. Cleaner, more editorial.
*   **Filled header:** header has a background that intensifies when expanded. High feedback.
*   **Nested accordion:** an accordion inside a panel (settings trees, category filters). Manage heading levels carefully (`<h3>` inside an `<h2>`\-headed item) and indentation.
*   **Icon-position variants:** chevron leading (left) reads as a tree/nav; chevron trailing (right) reads as an FAQ. Pick per context, be consistent.

## 7\. When to Use One (and When Not To)
Use an accordion when you have **peer sections of content the user scans by header and dips into selectively**, and showing them all at once would overwhelm or lengthen the page unhelpfully. FAQs, product spec groups, filter categories, multi-step forms (one step expanded at a time), settings sections, "what's included" breakdowns.

Don't use one when:

*   **The content is short enough to just show.** If all panels open fit on one screen without scrolling misery, the accordion is pure friction — you've added a click between the user and the answer. Especially true for FAQs where every answer is one sentence.
*   **Users need to compare sections side by side.** Single-open accordions actively fight comparison. Use multi-open, or don't use an accordion.
*   **It's mutually-exclusive views of one subject.** That's **Tabs** (see the sibling doc). Overview/Specs/Reviews is tabs; a list of independent FAQs is an accordion.
*   **It's primary navigation between pages.** That's a nav menu / disclosure nav, not a content accordion. (A collapsible _nav_ uses the disclosure pattern per item, not the exclusive-accordion pattern.)
*   **The content must be found via Ctrl-F or indexed as visible.** Collapsed content is easy to miss. If it's load-bearing for the user's task or your SEO, consider `hidden="until-found"` (native find-in-page reveals it) or don't collapse it at all.

Placement heuristics:

*   **Put the most-wanted section first and consider defaulting it open.** Don't make users hunt-and-click for the obvious answer.
*   **Keep headers scannable** — the header _is_ the summary. If a user can't predict the panel's contents from the header, the accordion is failing its one job.
*   **On mobile, accordions shine** (vertical stacking, thumb-reachable headers). This is why desktop tabs so often become mobile accordions.

## 8\. Accordions Across Design Systems
Same component, different philosophies and — crucially — different underlying engines.

*   **Material Design (Google):** the "Expansion Panel" / expandable list item. Elevation and a rotating chevron, ripple on the header, generous touch targets. Material 3 leans into shape and springy motion on expand.
*   **Apple HIG (iOS/macOS):** the disclosure group / disclosure triangle (the classic sideways triangle that rotates down). SwiftUI ships `DisclosureGroup`. Restrained, system-driven animation, tight to the platform.
*   **Fluent (Microsoft):** the Accordion in Fluent UI v9 — subtle, compact, `openItems`/`multiple` props, chevron indicators, built for dense productivity UIs.
*   **Ant Design / enterprise:** `Collapse` with `accordion` boolean for single-open, `Panel` children, ghost/borderless variants, `expandIcon` customization. Built for admin density and nested trees.
*   **Tailwind / utility-first:** no prescribed component — you compose it. Freedom plus the risk of shipping a `<div>`\-soup accordion with no semantics. Usually wrapped in a headless behavior layer to stay honest.
*   **shadcn / Radix (2024+ web standard):** `@radix-ui/react-accordion` handles all behavior and accessibility (roles, `aria-expanded`, keyboard, single/multiple); shadcn skins it with Tailwind + cva. The current default for React teams. Uses CSS custom properties (`--radix-accordion-content-height`) to solve the height-animation problem for you.
*   **Native HTML (****`<details>`****/****`<summary>`****):** the platform's own accordion primitive, no framework at all. With the 2024+ `name` attribute you get exclusive single-open behavior with _zero JavaScript_ (§9.1). The web-standards-purist choice, and increasingly the right default.

The through-line: every system gives you a **header trigger**, a **collapsible panel**, an **expanded-state indicator**, a **single vs multiple** switch, and a **keyboard contract**. Learn the concept once; remap the API per system.

## 8b. Visual Styles (the same accordion, eleven skins)
Design systems are _whose_ rules; visual styles are _which look_. The same semantic accordion — same headings, same `aria-expanded`, same keyboard map, same 44px targets — can wear any of these skins. Full guidance in the Design Styles (visual languages) gallery.

*   **Flat:** solid header fill, crisp hairline dividers, no shadow. Reads instantly.
*   **Material:** elevated item cards, ripple on the header, rotating chevron, springy expand.
*   **Glassmorphism:** translucent headers with `backdrop-filter: blur()` over a vivid backdrop. Guard the header text contrast hard.
*   **Liquid Glass (the 2026 trend):** Apple's iOS 26 / macOS Tahoe language. Headers as refractive translucent material with a specular rim and a sheen that shifts as panels open. Beautiful on deep backgrounds; honor reduced-motion for the sheen.
*   **Neumorphism:** headers as soft-extruded same-color surfaces, pressed-in when expanded. Pretty, contrast-poor — accent use only.
*   **Skeuomorphism:** beveled headers, inner highlights, a literal "folder tab" or ridged-panel metaphor. The tactile look.
*   **Neo-Brutalism:** thick borders, hard offset shadows, zero radius, a chunky +/− glyph. High personality FAQ.
*   **Claymorphism:** big radius, puffy headers with inner top-light, soft glow. Friendly, consumer.
*   **Aurora / Gradient:** animated multi-hue gradient behind the open panel or header. Premium; respect `prefers-reduced-motion`.
*   **Minimal / Swiss:** type and a single hairline rule do everything. Chevron trailing, generous whitespace, near-zero decoration.
*   **UJG Brand:** Goldenrod chevron and focus ring on deep Eminence/Night header surfaces, confident radius, a warm glow on the expanded item. The house default — the dark saturated ground makes an open panel feel lit from within.

**Rule that never changes:** the style is skin. Heading semantics, `aria-expanded`/`aria-controls`, the Enter/Space/Arrow keyboard contract, focus management, and the 44px header target stay identical across all eleven. Guard glass/liquid-glass/neu for contrast; guard aurora/liquid-glass for `prefers-reduced-motion`.

## 9\. The Code
### 9.1 HTML — native `<details>`/`<summary>` (master this first)
Before any framework, the browser already has an accordion. A single disclosure is one `<details>`. A _group_ of them is an accordion — and since 2024 you can make it **single-open with zero JavaScript** by giving every `<details>` the **same** **`name`** attribute.

```plain
<!-- A single disclosure: no JS, fully accessible, keyboard-operable for free -->
<details>
  <summary>What's your return policy?</summary>
  <div class="panel">30 days, no questions. We email a prepaid label.</div>
</details>

<!-- An EXCLUSIVE accordion: shared name = only one open at a time, still zero JS -->
<section class="accordion">
  <details name="faq">
    <summary>Shipping</summary>
    <div class="panel">Free over $50. 2–5 business days.</div>
  </details>
  <details name="faq">
    <summary>Returns</summary>
    <div class="panel">30 days for a full refund.</div>
  </details>
  <details name="faq" open>            <!-- `open` = expanded on load -->
    <summary>Warranty</summary>
    <div class="panel">One year, parts and labor.</div>
  </details>
</section>
```

What you get _for free_ with native `<details>`:

*   **Keyboard:** `<summary>` is focusable and toggles on **Enter** and **Space**. No JS.
*   **Semantics:** the browser exposes a disclosure/`aria-expanded` relationship to assistive tech automatically. The `open` attribute _is_ the state — no separate ARIA bookkeeping.
*   **Exclusive behavior:** matching `name` values make the group single-open, and — the good part — it **animates/reflows correctly** and even syncs across the group. This is the biggest quality-of-life addition to HTML accordions in a decade.
*   **Find-in-page:** browsers reveal a collapsed `<details>` when the user Ctrl-F's text inside it. Content stays findable even while visually collapsed — a real accessibility and SEO win the ARIA-button pattern doesn't get for free.

The catches (why teams still reach for the ARIA pattern):

*   **`<summary>`** **is a weird role**, not a plain button — deeply customizing its semantics (e.g. wrapping it in a heading correctly) takes care. Put the heading _inside_: `<summary><h3>Shipping</h3></summary>` is invalid nesting; instead style the summary as a heading or use `role="heading" aria-level="3"` on it.
*   **Animating to auto height** needs the new CSS (below) or a JS assist — the naive `<details>` snaps open.
*   **Fine-grained control** (analytics per toggle, "expand all," controlled state in React) is easier with the explicit ARIA pattern.

**Animating native** **`<details>`** **to auto height (2024/2025 CSS).** Historically impossible; now real. Two new primitives do it:

```css
/* 1) Opt into animating to intrinsic sizes (height:auto, etc.) */
:root { interpolate-size: allow-keywords; }

details::details-content {          /* the pseudo-element wrapping the panel */
  block-size: 0;
  overflow: clip;
  transition: block-size .3s ease, content-visibility .3s allow-discrete;
  content-visibility: hidden;
}
details[open]::details-content {
  block-size: auto;                 /* now animatable thanks to interpolate-size */
  content-visibility: visible;
}
```

`::details-content` targets the revealable content of a `<details>`, and `interpolate-size: allow-keywords` unlocks transitions _to_ keyword sizes like `auto`. Together they give a smooth open/close on a pure-HTML accordion. Where a browser doesn't support them yet, it degrades to an instant snap — still fully functional. **Progressive enhancement done right: it works everywhere, it's pretty where supported.**

### 9.2 HTML — the ARIA-button pattern (the explicit, controllable build)
When you need full control (controlled React state, analytics, "expand all," custom keyboard nav), build it from primitives per the WAI-ARIA Authoring Practices. The structure is exact and unforgiving:

```plain
<div class="accordion">
  <h3 class="accordion__header">
    <button type="button"
            id="acc-h1"
            class="accordion__trigger"
            aria-expanded="false"
            aria-controls="acc-p1">
      Shipping
      <svg class="chevron" aria-hidden="true" viewBox="0 0 20 20"><path d="M6 8l4 4 4-4"/></svg>
    </button>
  </h3>
  <div id="acc-p1"
       role="region"
       aria-labelledby="acc-h1"
       class="accordion__panel"
       hidden>
    <div class="accordion__inner">Free over $50. 2–5 business days.</div>
  </div>
  <!-- repeat: each item = heading > button + region -->
</div>
```

The non-negotiables, and why each exists:

*   **Header is a real heading** (`<h3>`) wrapping _only_ a `<button>`. The heading gives screen-reader users heading-navigation ("jump to next heading"); the button is the interactive part. Two elements, two jobs — this is the line juniors cross.
*   **`aria-expanded`** on the button reflects open/closed. This is _the_ state announcement. The rotating chevron is decorative (`aria-hidden`).
*   **`aria-controls`** points the button at the panel `id`; **`aria-labelledby`** points the panel back at the button `id`. Bidirectional wiring.
*   **`role="region"`** on the panel makes it a landmark a screen reader can jump to and announces its label. (For accordions with _many_ items, some teams drop `role="region"` to avoid landmark noise — a legitimate tradeoff; keep it for a handful, drop it for dozens.)
*   **`hidden`** collapses the panel — not `visibility` or clipping — so its focusable children leave the tab order. A link inside a "closed" panel that's still Tab-reachable is a classic bug.

### 9.4 The height-animation problem (the trick everyone gets wrong)
You **cannot** transition `height` from `0` to `auto`. CSS refuses to interpolate to an intrinsic keyword, so `transition: height .3s` with a target of `auto` just snaps. This one fact is behind 90% of janky accordions. There are three honest solutions, best-to-worst:

**Option A — the** **`grid-template-rows: 0fr → 1fr`** **trick (the modern default for hand-built accordions).** Wrap the panel content in a grid whose single row animates from `0fr` to `1fr`. `fr` units _are_ interpolatable, so the row grows smoothly to exactly the content's height with no JS measuring:

```css
.accordion__panel {
  display: grid;
  grid-template-rows: 0fr;                 /* collapsed */
  transition: grid-template-rows .3s ease;
}
.accordion__panel[data-open] {             /* or [aria-hidden="false"] */
  grid-template-rows: 1fr;                 /* expands to content height */
}
.accordion__panel > .accordion__inner {
  overflow: hidden;                        /* clips during the animation */
  min-height: 0;                           /* REQUIRED — grid children default to min-content */
}
```

The `min-height: 0` line is the gotcha — without it the grid child won't shrink below its content and nothing animates. This technique is pure CSS, needs no height measurement, and handles dynamic/lazy content automatically. It's why we don't put padding on `.accordion__panel` itself — padding would show even at `0fr`.

**Option B — the new native CSS (\*\*\*\*****`interpolate-size`** **/** **`::details-content`****\*\*\*\*).** Shown in §9.1. The cleanest when you're on native `<details>` and can target modern browsers. Degrades to a snap.

**Option C — JS measuring** **`scrollHeight`** **(the old reliable).** Set `height` to the measured `scrollHeight` in px, transition to it, then set to `auto` on `transitionend`; reverse on close. Works everywhere, but it's imperative, error-prone with dynamic content, and forces a layout read every toggle. Use only when you must support ancient browsers. Radix takes essentially this approach but hands you the measured value as `--radix-accordion-content-height`, so you write clean CSS against a variable it computes.

Whatever you pick: **animate only compositor-friendly work where possible, keep width fixed, and never animate** **`margin`****/****`top`** (they reflow every frame). And always wrap it in `prefers-reduced-motion`.

### 9.5 Vanilla JavaScript (the ARIA pattern, single & multi-open)
The behavior everything else abstracts. Worth knowing cold — including the optional Arrow-key roving between headers.

```javascript
class Accordion {
  constructor(root, { single = false } = {}) {
    this.root = root;
    this.single = single;                          // exclusive?
    this.triggers = [...root.querySelectorAll('.accordion__trigger')];

    this.triggers.forEach((btn) => {
      btn.addEventListener('click', () => this.toggle(btn));
      btn.addEventListener('keydown', (e) => this.onKeydown(e, btn));
    });
  }

  toggle(btn) {
    const expanded = btn.getAttribute('aria-expanded') === 'true';
    if (this.single && !expanded) {
      // exclusive: close the others first
      this.triggers.forEach((b) => b !== btn && this.setOpen(b, false));
    }
    this.setOpen(btn, !expanded);                  // allow-all-closed: clicking open closes it
  }

  setOpen(btn, open) {
    btn.setAttribute('aria-expanded', String(open));
    const panel = document.getElementById(btn.getAttribute('aria-controls'));
    panel.hidden = !open;                          // hidden removes focusables from tab order
    // if using the grid trick instead of [hidden]:
    // panel.toggleAttribute('data-open', open);
  }

  onKeydown(e, btn) {
    const i = this.triggers.indexOf(btn);
    let next;
    switch (e.key) {
      case 'ArrowDown': next = this.triggers[(i + 1) % this.triggers.length]; break;
      case 'ArrowUp':   next = this.triggers[(i - 1 + this.triggers.length) % this.triggers.length]; break;
      case 'Home':      next = this.triggers[0]; break;
      case 'End':       next = this.triggers[this.triggers.length - 1]; break;
      default: return;                             // let Enter/Space toggle natively
    }
    e.preventDefault();
    next.focus();                                  // Arrow keys move focus BETWEEN headers
  }
}

new Accordion(document.querySelector('.accordion'), { single: true });
```

Notes: **Enter and Space toggle** the focused header (native button behavior — don't reimplement it). **Arrow Up/Down move focus between headers** and are _optional but recommended_ per the ARIA APG; `Home`/`End` jump to first/last. Arrow keys move _focus_, they don't toggle — that's the accordion contract, distinct from a radio group where arrows also select.

### 9.6 React + TypeScript (a proper compound component)
The reusable, controlled/uncontrolled version. This is the longest block for a reason — it's the shape you'll actually ship. A **compound component** (Root/Item/Header/Panel sharing context) mirrors the DOM structure and keeps the API ergonomic.

```typescript
import {
  createContext, useContext, useId, useState, useCallback,
  type ReactNode, type Dispatch, type SetStateAction, type KeyboardEvent as ReactKeyboardEvent,
} from "react";

type AccordionMode = "single" | "multiple";

interface AccordionCtx {
  openIds: Set<string>;
  toggle: (id: string) => void;
  mode: AccordionMode;
}
const Ctx = createContext<AccordionCtx | null>(null);
const useAccordion = () => {
  const c = useContext(Ctx);
  if (!c) throw new Error("Accordion.* must be used inside <Accordion>");
  return c;
};

interface RootProps {
  mode?: AccordionMode;
  defaultOpen?: string[];
  collapsible?: boolean;          // single-mode: allow closing the open one
  children: ReactNode;
}

export function Accordion({ mode = "single", defaultOpen = [], collapsible = true, children }: RootProps) {
  const [openIds, setOpenIds] = useState<Set<string>>(new Set(defaultOpen));

  const toggle = useCallback((id: string) => {
    setOpenIds((prev) => {
      const next = new Set(prev);
      const isOpen = next.has(id);
      if (mode === "single") {
        next.clear();
        if (!isOpen) next.add(id);
        else if (!collapsible) next.add(id);   // not collapsible => keep it open
      } else {
        isOpen ? next.delete(id) : next.add(id);
      }
      return next;
    });
  }, [mode, collapsible]);

  return <Ctx.Provider value={{ openIds, toggle, mode }}><div className="accordion">{children}</div></Ctx.Provider>;
}

interface ItemCtx { headerId: string; panelId: string; id: string; open: boolean; }
const ItemContext = createContext<ItemCtx | null>(null);
const useItem = () => {
  const c = useContext(ItemContext);
  if (!c) throw new Error("Accordion.Header/Panel must be inside Accordion.Item");
  return c;
};

export function AccordionItem({ id, children }: { id?: string; children: ReactNode }) {
  const auto = useId();
  const itemId = id ?? auto;
  const { openIds } = useAccordion();
  const value: ItemCtx = {
    id: itemId,
    headerId: `${itemId}-header`,
    panelId: `${itemId}-panel`,
    open: openIds.has(itemId),
  };
  return <ItemContext.Provider value={value}>{children}</ItemContext.Provider>;
}

export function AccordionHeader({ children, level = 3 }: { children: ReactNode; level?: 2 | 3 | 4 }) {
  const { toggle } = useAccordion();
  const { id, headerId, panelId, open } = useItem();
  const Heading = `h${level}` as const;

  // Arrow Up/Down + Home/End rove focus BETWEEN headers (APG-recommended). Enter/Space
  // fall through to native <button> activation — we never preventDefault them.
  const onKey = (e: ReactKeyboardEvent<HTMLButtonElement>) => {
    const root = e.currentTarget.closest(".accordion");
    if (!root) return;
    const triggers = [...root.querySelectorAll<HTMLButtonElement>(".accordion__trigger")];
    const i = triggers.indexOf(e.currentTarget);
    let next: HTMLButtonElement | undefined;
    switch (e.key) {
      case "ArrowDown": next = triggers[(i + 1) % triggers.length]; break;
      case "ArrowUp":   next = triggers[(i - 1 + triggers.length) % triggers.length]; break;
      case "Home":      next = triggers[0]; break;
      case "End":       next = triggers[triggers.length - 1]; break;
      default: return;                        // Enter/Space toggle via the native button
    }
    e.preventDefault();
    next?.focus();
  };

  return (
    <Heading className="accordion__header">
      <button
        type="button"
        id={headerId}
        className="accordion__trigger"
        aria-expanded={open}
        aria-controls={panelId}
        onClick={() => toggle(id)}
        onKeyDown={onKey}
      >
        <span>{children}</span>
        <svg className="chevron" aria-hidden viewBox="0 0 20 20"><path d="M6 8l4 4 4-4" /></svg>
      </button>
    </Heading>
  );
}

export function AccordionPanel({ children }: { children: ReactNode }) {
  const { headerId, panelId, open } = useItem();
  return (
    <div
      id={panelId}
      role="region"
      aria-labelledby={headerId}
      className="accordion__panel"
      hidden={!open}          // hidden pulls focusables out of tab order
    >
      <div className="accordion__inner">{children}</div>
    </div>
  );
}
```

Usage reads like the markup it produces:

```plain
<Accordion mode="single" collapsible defaultOpen={["shipping"]}>
  <AccordionItem id="shipping">
    <AccordionHeader>Shipping</AccordionHeader>
    <AccordionPanel>Free over $50. 2–5 business days.</AccordionPanel>
  </AccordionItem>
  <AccordionItem id="returns">
    <AccordionHeader>Returns</AccordionHeader>
    <AccordionPanel>30 days for a full refund.</AccordionPanel>
  </AccordionItem>
</Accordion>
```

Why this is the senior version, not the junior one:

*   **Compound components + context** mean the consumer writes structure, not config objects. Each part reads shared state; no prop-drilling.
*   **`useId`** generates collision-proof, SSR-safe `id`s for the `aria-controls`/`aria-labelledby` wiring — the #1 thing hand-rolled accordions botch.
*   **`mode`** **+** **`collapsible`** cover single-exclusive, single-with-close, and multi-open from one component.
*   **`hidden={!open}`** (not CSS-only hiding) keeps collapsed panels out of the tab order automatically.
*   **The** **`level`** **prop** forces the caller to place the item in the correct heading hierarchy instead of hardcoding `<h3>` everywhere.

For animation, swap `hidden` for the grid-rows trick (§9.4) driven by `data-open={open || undefined}`, so closing animates instead of snapping. Guard `prefers-reduced-motion`.

### 9.8 shadcn / Radix (the 2024+ React standard)
You don't install a black box — you copy shadcn's Accordion into your repo; **Radix** supplies behavior + accessibility, **cva/Tailwind** supply looks. Radix solves the height animation by exposing the measured height as a CSS variable.

```plain
"use client";
import * as React from "react";
import * as AccordionPrimitive from "@radix-ui/react-accordion";
import { ChevronDown } from "lucide-react";        // named import — tree-shakeable
import { cn } from "@/lib/utils";

const Accordion = AccordionPrimitive.Root;          // type="single" | "multiple", collapsible

const AccordionItem = React.forwardRef<
  React.ElementRef<typeof AccordionPrimitive.Item>,
  React.ComponentPropsWithoutRef<typeof AccordionPrimitive.Item>
>(({ className, ...props }, ref) => (
  <AccordionPrimitive.Item ref={ref} className={cn("border-b border-white/10", className)} {...props} />
));
AccordionItem.displayName = "AccordionItem";

const AccordionTrigger = React.forwardRef<
  React.ElementRef<typeof AccordionPrimitive.Trigger>,
  React.ComponentPropsWithoutRef<typeof AccordionPrimitive.Trigger>
>(({ className, children, ...props }, ref) => (
  <AccordionPrimitive.Header className="flex">     {/* renders a real heading */}
    <AccordionPrimitive.Trigger
      ref={ref}
      className={cn(
        "flex flex-1 items-center justify-between gap-3 min-h-11 py-4 text-sm font-semibold",
        "transition-all hover:underline focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-amber-400",
        "[&[data-state=open]>svg]:rotate-180", className
      )}
      {...props}
    >
      {children}
      <ChevronDown className="size-4 shrink-0 transition-transform duration-200" aria-hidden />
    </AccordionPrimitive.Trigger>

));
AccordionTrigger.displayName = AccordionPrimitive.Trigger.displayName;

const AccordionContent = React.forwardRef<
  React.ElementRef<typeof AccordionPrimitive.Content>,
  React.ComponentPropsWithoutRef<typeof AccordionPrimitive.Content>
>(({ className, children, ...props }, ref) => (
  <AccordionPrimitive.Content
    ref={ref}
    className={cn(
      // Radix gives you --radix-accordion-content-height; animate against it
      "overflow-hidden text-sm data-[state=closed]:animate-accordion-up data-[state=open]:animate-accordion-down",

    )}
    {...props}

```

With the keyframes in your Tailwind config:

```plain
keyframes: {
  "accordion-down": { from: { height: "0" }, to: { height: "var(--radix-accordion-content-height)" } },
  "accordion-up":   { from: { height: "var(--radix-accordion-content-height)" }, to: { height: "0" } },
},

```

Usage: `<Accordion type="single" collapsible>`. The clever part is `--radix-accordion-content-height` — Radix measures the panel and hands you the pixel value, so you animate real `height` cleanly without writing the `scrollHeight` JS yourself. `type="multiple"` switches to multi-open; `collapsible` lets single-mode close the open item. All roles, `aria-expanded`, keyboard, and focus are handled by the primitive.

### 9.9 Next.js (App Router: client island, server content)
An accordion that _toggles_ needs `onClick`/state, so the interactive shell is a **Client Component** — but the panel _content_ can be rendered on the server and passed in as `children`, keeping the payload lean and letting server data flow in without an API call.

```plain
// app/faq/Faq.tsx  — the interactive shell
"use client";
import { Accordion, AccordionItem, AccordionHeader, AccordionPanel } from "@/components/accordion";

export function Faq({ items }: { items: { id: string; q: string; a: React.ReactNode }[] }) {
  return (
    <Accordion mode="single" collapsible>
      {items.map(({ id, q, a }) => (
        <AccordionItem key={id} id={id}>
          <AccordionHeader>{q}</AccordionHeader>
          <AccordionPanel>{a}</AccordionPanel>   {/* `a` was rendered on the server */}
        </AccordionItem>
      ))}
    </Accordion>
  );
}

// app/faq/page.tsx — Server Component: fetch on the server, hand content down
import { Faq } from "./Faq";
export default async function Page() {
  const faqs = await getFaqs();               // runs on the server, no client fetch
  return <Faq items={faqs} />;
}
```

The distinction that trips people up: **interactivity is the client boundary, data is not.** You don't need a client-side fetch for accordion content — render it server-side and pass it as `children`/props into the small client island that owns the open/closed state. For _lazy_ panels (fetch on expand), the panel body becomes its own client component that fires the request in an effect keyed off `open`, with a skeleton state (§5).

### 9.10 Deep-linking (open a section from the URL)
An accordion whose open section is driven by the URL hash is shareable, back-button-friendly, and SEO-visible. Click a link to `/faq#returns` and the Returns panel opens and scrolls into view.

```plain
"use client";
import { useEffect, useState } from "react";

function useHashOpen(defaultId?: string) {
  const [openId, setOpenId] = useState(defaultId);
  useEffect(() => {
    const apply = () => {
      const id = location.hash.slice(1);
      if (id) {
        setOpenId(id);
        // let the panel expand, then bring it into view
        requestAnimationFrame(() => document.getElementById(id)?.scrollIntoView({ block: "start" }));
      }
    };
    apply();                                   // on mount (direct link / refresh)
    window.addEventListener("hashchange", apply);
    return () => window.removeEventListener("hashchange", apply);
  }, []);
  return [openId, setOpenId] as const;
}
```

Native `<details>` gets a slice of this **for free**: browsers now auto-expand a `<details>` when you navigate to a fragment pointing at content inside it (fragment-directive / `hidden=until-found` behavior). For the ARIA-button pattern you wire it yourself as above. Either way: **sync open state to the URL** for FAQ-style accordions — it's a real UX and SEO upgrade, and near-zero cost.

### 9.11 Testing the Accordion (behavior + a11y)
An accordion is behavior, so test behavior, not class names. Query by role and by _accessible state_.

```typescript
import { render, screen } from "@testing-library/react";
import userEvent from "@testing-library/user-event";
import { axe } from "jest-axe";
import { describe, it, expect } from "vitest";
import { Faq } from "./Faq";

const items = [
  { id: "a", q: "Shipping", a: "Free over $50" },
  { id: "b", q: "Returns", a: "30 days" },
];

describe("Accordion", () => {
  it("exposes each header as a button with expanded state", () => {
    render(<Faq items={items} />);
    const shipping = screen.getByRole("button", { name: /shipping/i });
    expect(shipping).toHaveAttribute("aria-expanded", "false");
  });

  it("expands on click and reveals the panel", async () => {
    render(<Faq items={items} />);
    const shipping = screen.getByRole("button", { name: /shipping/i });
    await userEvent.click(shipping);
    expect(shipping).toHaveAttribute("aria-expanded", "true");
    expect(screen.getByRole("region", { name: /shipping/i })).toBeVisible();
  });

  it("single mode closes the previously open section", async () => {
    render(<Faq items={items} />);
    await userEvent.click(screen.getByRole("button", { name: /shipping/i }));
    await userEvent.click(screen.getByRole("button", { name: /returns/i }));
    expect(screen.getByRole("button", { name: /shipping/i })).toHaveAttribute("aria-expanded", "false");
    expect(screen.getByRole("button", { name: /returns/i })).toHaveAttribute("aria-expanded", "true");
  });

  it("is keyboard operable (Enter toggles, Arrow moves between headers)", async () => {
    render(<Faq items={items} />);
    await userEvent.tab();                               // focus first header
    await userEvent.keyboard("{Enter}");
    expect(screen.getByRole("button", { name: /shipping/i })).toHaveAttribute("aria-expanded", "true");
    await userEvent.keyboard("{ArrowDown}");
    expect(screen.getByRole("button", { name: /returns/i })).toHaveFocus();
  });

  it("has no axe violations in either state", async () => {
    const { container } = render(<Faq items={items} />);
    expect(await axe(container)).toHaveNoViolations();
    await userEvent.click(screen.getByRole("button", { name: /shipping/i }));
    expect(await axe(container)).toHaveNoViolations();   // re-check while open
  });
});
```

What to assert (and skip):

*   **Do test:** header is found `byRole("button")`, `aria-expanded` flips, the panel is a `region` labeled by its header, single-mode closes siblings, Enter/Space toggles, Arrow moves focus, collapsed panels remove their content/focusables. Run `axe` in _both_ open and closed states — bugs hide in one.
*   **Don't test:** exact chevron rotation degrees, hex colors, class names, animation duration. Implementation details that break tests without catching real bugs.
*   **E2E (Playwright):** assert the panel's focusable children are _not_ tabbable when collapsed, and that a deep-link (`/faq#returns`) opens and scrolls to the right panel.

### 9.14 Web Component (framework-agnostic, native `<details>` inside)
A portable accordion item that leans on `<details>` for free behavior and only adds animation + styling:

```typescript
class UiAccordionItem extends HTMLElement {
  connectedCallback() {
    // Attributes are untrusted (CMS/user-authored). Escape before they touch innerHTML,
    // or title/name become an XSS / attribute-injection sink — the exact §15 rule.
    const esc = (s: string) =>
      s.replace(/[&<>"']/g, (c) =>
        ({ "&": "&", "<": "<", ">": ">", '"': """, "'": "'" })[c] as string);
    const title = esc(this.getAttribute("title") ?? "");
    const openAttr = this.hasAttribute("open") ? "open" : "";
    const name = this.getAttribute("name");                 // shared name => exclusive group
    this.attachShadow({ mode: "open" }).innerHTML = `
      <style>
        details { border-bottom: 1px solid rgba(255,255,255,.1); }
        summary { list-style: none; cursor: pointer; min-height: 44px;
                  display: flex; align-items: center; justify-content: space-between;
                  padding: 12px 16px; font: 600 15px system-ui; }
        summary:focus-visible { outline: 2px solid #DCA424; outline-offset: -2px; }
        .chevron { transition: transform .2s; }
        details[open] .chevron { transform: rotate(180deg); }
        .inner { padding: 16px; }
        @media (prefers-reduced-motion: reduce) { .chevron { transition: none; } }
      </style>
      <details ${name ? `name="${esc(name)}"` : ""} ${openAttr}>
        <summary>${title}<span class="chevron" aria-hidden>▾</span></summary>
        <div class="inner"><slot></slot></div>
      </details>`;
  }
}
customElements.define("ui-accordion-item", UiAccordionItem);
// Usage: <ui-accordion-item name="faq" title="Shipping">Free over $50</ui-accordion-item>
```

Passing the same `name` to several items gives you exclusive single-open behavior with no JS state machine at all — the native `name` attribute doing the work inside the shadow root.

## 10\. Accessibility (non-negotiable)
The accordion is one of the densest a11y contracts in UI. Every point here has caused a real audit failure.

**Structure & roles:**

*   **Each header is a real heading** (`<h2>`/`<h3>`, correct level for the page outline) wrapping a **native** **`<button>`**. Not a styled `<div>`. The heading enables heading-navigation; the button enables activation.
*   **The button carries** **`aria-expanded`** (`true`/`false`) — this _is_ the state announcement. And **`aria-controls`** pointing at the panel `id`.
*   **The panel is** **`role="region"`** **with** **`aria-labelledby`** pointing back at its button, so entering it announces "Shipping, region." (Drop `role="region"` for accordions with _many_ items to avoid landmark spam — a documented tradeoff.)
*   **Native** **`<details>`****/****`<summary>`** gives most of this automatically; the main care is making `<summary>` read as a heading and animating without breaking it.

**Keyboard map (the contract):**

| Key | Action |
| ---| --- |
| Enter / Space | Toggle the focused header open/closed |
| Tab | Move to the next focusable (header, or focusable inside an open panel) |
| Down / Up Arrow | (Optional, recommended) Move focus between headers |
| Home / End | (Optional) Move focus to first / last header |

*   **Arrow keys move focus, they don't toggle** — that's what separates an accordion from a radio group. And they're optional: the ARIA APG lists them as recommended, not required. Enter/Space on the header is the required minimum.
*   **Collapsed panels must remove their focusable children from the tab order** — use `hidden`/`display:none`, never opacity or clipping. A Tab that lands on an invisible link is a WCAG 2.1.1/2.4.3 failure.
*   **Focus survives a collapse:** if the user collapses the section their focus is inside, move focus to that section's header — never leave focus on a now-hidden element.

**Perception:**

*   **Never signal state by icon alone.** The chevron is `aria-hidden`; `aria-expanded` does the announcing. Don't rely on color alone either.
*   **Contrast ≥ 4.5:1** for header text; the focus ring must be visible against the header background (mind glass/neumorphism skins).
*   **Header tap target ≥ 44px**, full-width-clickable — the whole header row, not just the label or the chevron.
*   **Respect** **`prefers-reduced-motion`****\*\*\*\*:** kill the height/chevron transition, snap instead.

## 11\. Innovative & Emerging Ideas (2026-aware)
The accordion looks solved but is quietly getting better:

*   **Native height animation is finally here.** `interpolate-size: allow-keywords` + `::details-content` + `transition-behavior: allow-discrete` mean pure-CSS `<details>` accordions now animate to `auto` height with no JS. This is the single biggest shift — the grid-rows hack becomes a fallback, not the default, as support lands.
*   **Exclusive accordions with zero JS** via the `<details name>` attribute — shipped across modern browsers in 2024–2025. Single-open behavior is now a platform feature, not a state machine.
*   **`hidden="until-found"`** **\+ find-in-page:** collapsed content that stays Ctrl-F-findable and auto-expands when matched — closing the "hidden content is lost content" gap that always plagued accordions.
*   **Scroll-driven and view-transition reveals:** the View Transitions API morphs a collapsing panel smoothly across state, and scroll-timeline lets panels progressively reveal — richer than a height tween, still declarative.
*   **AI-summarized headers:** panels whose header text is a model-generated one-line summary of the content inside, so the "is the header a good summary?" bet (§2) is made automatically and stays accurate as content changes.
*   **Anchor-positioned nested reveals** and container-query-driven density: accordions that switch compact/comfortable based on their _container_ width, not the viewport — the right primitive for component-level responsiveness.

**For UJG:** default new FAQ/spec accordions to native `<details name>` with the `interpolate-size` animation and a Goldenrod focus ring on Eminence headers — it's the least code, the most accessible, and the most on-trend. Reserve the JS/Radix build for accordions that need controlled state, analytics per toggle, or lazy-loaded panels.

## 12\. Conversion & UX Killers (mistakes that bleed trust silently)
None of these throw an error. They just quietly hurt your numbers and your audits.

*   **Hiding the answer people came for.** Collapsing your pricing, return policy, or key spec means most users never see it. If it's the thing they want, default it open or don't collapse it. Every collapse is a bet the header summarizes well enough — lose that bet and you lose the user.
*   **The** **`<div onClick>`** **header.** No heading, no `aria-expanded`, no keyboard. Looks fine, is invisible to screen readers and unusable by keyboard. The most common accordion failure in the wild, and an instant audit fail.
*   **No open/close animation — or a janky one.** Instant snap feels broken; a height animation that reflows and jumps at the end (padding on the animated element, §9.4) feels amateur. Perceived quality lives here.
*   **Layout shift (CLS) on expand.** Panels that shove content below them without reserving space tank Core Web Vitals and feel unstable. Animate height smoothly; don't let the page lurch.
*   **Everything collapsed by default with vague headers.** A wall of identical closed rows with headers like "Details" or "More info" gives the user nothing to decide on. Headers must predict their panels.
*   **Single-open when users need to compare.** Forcing exclusive mode on a spec comparison means every click hides what they just read. Use multi-open when comparison is the job.
*   **Chevron-only click target.** Making only the tiny caret clickable instead of the whole 44px header row = missed taps and rage. The entire header is the button.
*   **Focus lost on collapse.** Collapse the section the user's focus is in without moving focus, and keyboard/AT users are dumped to the top of the document. Move focus to the header.

**The through-line:** these fail _quietly_ — no crash, just a lower engagement number and a failed audit that's easy to blame on "the content." Audit your accordions before you rewrite your copy.

## 13\. Advanced Patterns
Senior-move versions of the patterns above.

**Expand-all / collapse-all controls.** For long multi-open accordions, a pair of buttons that set every item's state at once. Wire them to the same state store the items read; announce the bulk change via an `aria-live` region ("All sections expanded") so AT users aren't left guessing.

**Persisted open state.** Remember which sections a user opened across visits (localStorage keyed by item id, or URL hash per §9.10). Respect it on load — but never let a stale persisted state hide _new_, important content.

**TypeScript-enforced heading levels.** Force the caller to pass a valid heading level so the accordion can't silently break the document outline:

```typescript
type HeadingLevel = 2 | 3 | 4 | 5 | 6;
interface AccordionHeaderProps { level: HeadingLevel; children: React.ReactNode; }
// <AccordionHeader> without a level won't compile — the outline stays correct by construction.
```

**Design tokens at scale.** The accordion references _component_ tokens that reference _semantic_ tokens that reference _primitives_, so a rebrand or dark-mode is a token swap, not a rewrite:

```css
:root {
  --purple-800: oklch(42% 0.145 305);   /* primitive */
  --gold-500:   oklch(78% 0.135 82);
  --color-surface: var(--purple-800);   /* semantic */
  --color-focus:   var(--gold-500);
  --acc-header-bg: color-mix(in oklch, var(--color-surface), black 55%);  /* component */
  --acc-ring:      var(--color-focus);
}
.accordion__trigger { background: var(--acc-header-bg); }
.accordion__trigger:focus-visible { outline: 2px solid var(--acc-ring); }
```

**Lazy panels with a stable header.** Fetch panel content on first expand, cache it, show a skeleton meanwhile, and keep the header instantly responsive (open the empty animated shell immediately; stream content in). Never block the toggle on the network.

**aria-live for async panel results.** When a lazy panel finishes loading or errors, announce it politely so screen-reader users learn the outcome without the visual spinner:

```plain
<div id="p1" role="region" aria-labelledby="h1">
  <div class="accordion__inner" aria-live="polite"><!-- skeleton → content / "Couldn't load, retry" --></div>
</div>
```

## 14\. Performance & Bundle Cost
An accordion feels free. At scale it isn't.

*   **The icon-import trap.** Importing a whole icon library for one chevron is the most common accordion perf mistake. `import { ChevronDown } from "lucide-react"` (tree-shakeable) — never `import * as Icons`, which can add hundreds of KB.
*   **Don't thrash layout.** Animating `height`/`margin`/`top` reflows every frame. Prefer the compositor-friendly path where you can, and where you must animate height, use the grid-rows `fr` trick or Radix's measured variable rather than animating `margin`. Keep width fixed.
*   **`content-visibility: auto`** on collapsed panels lets the browser skip rendering their offscreen/hidden content entirely — a real win for accordions with dozens of heavy panels. Pair with `contain-intrinsic-size` so the scrollbar doesn't jump.
*   **Don't mount closed-panel content you don't need.** For an accordion with 100 items, render panel bodies on demand (mount on first open) rather than shipping all 100 to the DOM up front. Balance against SEO — if the content must be indexed, render it (collapsed) but keep it light.
*   **One delegated listener** on the accordion root beats 50 per-header listeners for large lists — fewer closures, and it covers dynamically-added items (§9.5's class can be adapted to delegation).
*   **In React,** memoize item components and wrap toggles in `useCallback` so opening one section doesn't re-render all hundred. Don't over-apply — reach for it when the list is genuinely large.

## 15\. Security
An accordion's own machinery — a button, a labeled region, a height animation — carries almost no attack surface. The risk lives in what you render _inside_ the panels, so treat the body as untrusted content, not chrome.

*   **Panel bodies are an injection sink.** FAQ answers, help text, and CMS- or user-authored copy routinely land in a panel. Push them through `innerHTML` and you've built a stored-XSS vector that fires the moment someone expands the item — render as text, or sanitize (DOMPurify / a server-side allow-list) before it touches the DOM, never raw HTML.
*   **Lazy panels inherit real authorization duty.** A panel that fetches its body from an endpoint on first open is a network call like any other: that endpoint owns the authz check and the output-escaping, and a collapsed panel is _not_ an access control. Don't ship gated content to the client "hidden" behind a closed panel — anyone can read it in the DOM or the network tab.
*   **Deep-linking must respect authorization.** A URL hash or query param that auto-expands a specific panel must not surface — or auto-open — content the viewer isn't cleared to see; gate on identity server-side, not on whether the panel happens to be closed.

Beyond those three, an accordion adds nothing new: it's a disclosure widget, and its security is the security of its contents. Get the sanitization and the "closed ≠ protected" instinct right and the component is covered.

## 16\. Senior-Level Checklist (ship-ready)
Before an accordion is "done":

- [ ] Each header is a real heading (`<h2>`/`<h3>`, correct level) wrapping a native `<button>` — never a `<div>`.
- [ ] Button has `aria-expanded` and `aria-controls`; panel has `role="region"` (or intentionally omitted for many items) and `aria-labelledby`.
- [ ] `id` wiring is collision-proof (`useId`/UUID), verified in both directions.
- [ ] Collapsed panels use `hidden`/`display:none` — focusable children leave the tab order.
- [ ] Enter/Space toggles; Arrow Up/Down move focus between headers (if implemented); focus survives a collapse.
- [ ] Height animates smoothly (grid-rows `fr`, native `interpolate-size`, or Radix var) — no snap, no end-jump; padding on the inner wrapper, not the animated element.
- [ ] `prefers-reduced-motion` kills the transition.
- [ ] Icon rotation is driven off `aria-expanded`/`data-state`, not a separately-toggled class.
- [ ] Single vs multi-open behavior is intentional; single-open allows "all closed."
- [ ] Header tap target ≥ 44px, full row clickable; header text contrast ≥ 4.5:1; focus ring visible on the header skin.
- [ ] No layout shift (CLS) on expand; lazy panels show skeleton + error + retry.
- [ ] Deep-linkable open state where it helps (FAQs); important content isn't buried by default.
- [ ] `axe` passes in _both_ open and closed states.

## 17\. Visual Styles Reference
The same accordion — same heading structure, same `aria-expanded`, same keyboard contract, same 44px header — rendered across the eleven visual languages. **Style is skin; behavior is the skeleton.**

*   **Flat:** solid header fill, hairline dividers, no shadow, chevron trailing. The default FAQ look; scales fast.
*   **Material:** each item an elevated card, ripple on the header, chevron rotates, springy Material-3 expand.
*   **Glassmorphism:** translucent blurred headers over a vivid backdrop — guard header text contrast, keep a hairline so items don't dissolve.
*   **Liquid Glass (2026):** refractive translucent headers with a specular rim; the open panel's surface picks up a subtle sheen. Deep backgrounds make it sing; reduced-motion disables the sheen.
*   **Neumorphism:** soft-extruded same-color headers, pressed-in when expanded. Contrast-poor — accent only, and test the focus ring hard.
*   **Skeuomorphism:** beveled "folder tab" headers, ridged panel texture, inner highlights. The tactile metaphor.
*   **Neo-Brutalism:** thick black borders, hard offset shadow on the open item, zero radius, a chunky +/− glyph instead of a chevron.
*   **Claymorphism:** big radius, puffy headers with inner top-light and soft glow. Friendly, consumer FAQ.
*   **Aurora / Gradient:** animated multi-hue gradient behind the expanded panel or the active header. Premium; honor `prefers-reduced-motion`.
*   **Minimal / Swiss:** a single hairline rule and type do everything, chevron trailing, generous whitespace, sharp corners.
*   **UJG Brand:** Goldenrod chevron and focus ring on deep Eminence/Night header surfaces, confident radius, a warm glow lighting the open item from within. The house default.

Full cross-reference in the Design Styles (visual languages) gallery. Every style skins every mode (single/multi) from one shared token set — the guard rails (contrast for glass/neu, reduced-motion for aurora/liquid-glass) never change.

**Bottom line:** an accordion is a deceptively deep component — a heading, a button, a controlled region, a keyboard contract, a focus-management problem, and a height-animation puzzle, all in one repeating unit. The two engines (native `<details name>` with `interpolate-size`, or the explicit heading + `aria-expanded` button + labeled region) are both correct; pick native for the least code and the most accessibility-for-free, pick the ARIA/Radix build when you need controlled state, analytics, or lazy panels. Nail the `aria-expanded` wiring, the grid-rows-or-native height animation, the keyboard map, and the "don't bury the answer" judgment call, and you've mastered a component most engineers ship broken. Read it alongside its sibling **Tabs** — same instinct to show one thing at a time, opposite layout and opposite contract.
