        type ReactElement, type ReactNode,
      } from "react";
      import { createPortal } from "react-dom";

      interface TooltipProps {
        label: ReactNode;              // SHORT, non-interactive text
        children: ReactElement;        // exactly one focusable trigger
        openDelay?: number;
        closeDelay?: number;
        placement?: "top" | "bottom" | "start" | "end";
      }

      export function Tooltip({
        label, children, openDelay = 400, closeDelay = 150, placement = "top",
      }: TooltipProps) {
        const id = useId();
        const [open, setOpen] = useState(false);
        const openT = useRef<number>();
        const closeT = useRef<number>();

        const show = (delay: number) => {
          window.clearTimeout(closeT.current);
          openT.current = window.setTimeout(() => setOpen(true), delay);
        };
        const hide = () => {
          window.clearTimeout(openT.current);
          closeT.current = window.setTimeout(() => setOpen(false), closeDelay);
        };

        useEffect(() => {
          if (!open) return;
          const onKey = (e: KeyboardEvent) => { if (e.key === "Escape") setOpen(false); };
          document.addEventListener("keydown", onKey);
          return () => document.removeEventListener("keydown", onKey);
        }, [open]);

        // Clear any pending open/close timer on unmount, so a tooltip that unmounts
        // mid-delay (e.g. a row that scrolls away) can't fire setOpen after teardown
        // — the classic "state update on an unmounted component" leak.
        useEffect(() => () => {
          window.clearTimeout(openT.current);
          window.clearTimeout(closeT.current);
        }, []);

        if (!isValidElement(children)) {
          throw new Error("Tooltip expects a single element child as the trigger.");
        }

        // Merge our handlers + the describedby onto the REAL trigger element.
        const trigger = cloneElement(children, {
          "aria-describedby": open ? id : undefined,
          onPointerEnter: () => show(openDelay),
          onPointerLeave: hide,
          onFocus: () => show(0),          // focus = instant, no delay
          onBlur: () => setOpen(false),
        });

        return (
          <>
            {trigger}
            {open &&
              createPortal(
                <div
                  role="tooltip"
                  id={id}
                  data-placement={placement}
                  className="tooltip"
                  onPointerEnter={() => window.clearTimeout(closeT.current)}  // hoverable
                  onPointerLeave={hide}
                >
                  {label}
                </div>,
                document.body,
              )}
          </>
        );
      }

## 1\. What a Tooltip Actually Is
A tooltip is a small, **non-interactive** text bubble that gives a _supplemental label or hint_ about an element the user is already pointing at or has focused. That's it. Two words carry the entire definition: **supplemental** and **non-interactive**.

* **Supplemental** means the interface still works if the tooltip never appears. It clarifies or names — it never delivers the only copy of something essential. If the app breaks without it, it isn't a tooltip.
* **Non-interactive** means there is nothing inside it to click, tab to, select, or read line-by-line. No links, no buttons, no form fields, no "click here." The moment content becomes interactive, you no longer have a tooltip.

The single most common conceptual error in UI engineering is lumping four different things under the word "tooltip." Get these distinctions right and 80% of tooltip bugs disappear:

* **Tooltip:** appears on **hover OR keyboard focus**, describes an _already-labelled_ control, non-interactive, dismissable. `role="tooltip"` + `aria-describedby`. Example: hovering an icon-only "trash" button and seeing "Delete item."
* **Toggletip:** revealed by a **click/tap on a button**, not hover. Used when the extra info is a genuine aside the user chooses to open ("What's a routing number?"). Announced via an `aria-live` region, _not_ `aria-describedby`. It's a button that reveals a bubble, semantically closer to a disclosure.
* **Popover:** an **interactive** overlay — it can contain links, buttons, forms, rich content. It takes focus. This is a menu, a date picker, a rich card. Use the [Popover API](https://developer.mozilla.org/en-US/docs/Web/API/Popover_API) / a dialog, never `role="tooltip"`.
* **`title`** **attribute:** the browser's built-in native tooltip. It is _not_ a real solution — it's unreliable, and we spend a whole section (§9.6) on why you should almost never ship it.

The distinction people get wrong most: **tooltip vs. toggletip.** If the trigger is a control the user _hovers_ to clarify what it does, it's a tooltip and it describes. If the trigger is a control the user _clicks_ to reveal a side note, it's a toggletip and it announces. Hover to describe; click to reveal. Mixing the two ARIA patterns is the #1 tell of a build that copied a snippet without understanding it.

The second distinction people get wrong: **describes vs. labels.** A tooltip _describes_ (`aria-describedby`) — it's secondary info attached to a control that already has a name. It does **not** _label_ (`aria-labelledby`) an unlabelled control. If your icon button has no accessible name and you're using the tooltip to _be_ the name, you've built it wrong — the name belongs in `aria-label` on the button; the tooltip is bonus.

## 2\. Why It Matters
A tooltip feels trivial, so teams ship it carelessly — which is exactly why it does disproportionate damage.

* **It's an accessibility trap, not a freebie.** Unlike a button (where a real `<button>` hands you correctness for free), the tooltip has _no_ safe native primitive. Every mistake — hover-only, essential content, interactive content, no Esc, no focus trigger — is an active WCAG failure (2.1.1 Keyboard, 1.4.13 Content on Hover or Focus, 4.1.2 Name/Role/Value). Tooltips are one of the most-cited failure categories in real accessibility audits.
* **It's where teams hide information they should have surfaced.** The tooltip becomes a junk drawer: critical instructions, error explanations, form requirements, pricing caveats — all buried behind a hover that mobile users and keyboard users and screen-reader users may never trigger. That's not a styling bug; it's information architecture failing silently.
* **It carries trust.** A tooltip that appears instantly on every mouse-over (no delay), flickers, gets clipped at the viewport edge, or vanishes the moment you try to read it makes an interface feel amateur and jittery. Smooth, well-delayed, collision-aware tooltips are a quiet signal of craft.

For UJG client work the stakes are concrete: a form where the "why do we need this?" hint only exists in a hover tooltip will lose mobile conversions and can surface as a legal accessibility exposure. Tooltips are cheap to build and expensive to get wrong.

## 3\. Anatomy of a Tooltip
Every well-built tooltip is composed of predictable parts:

* **Trigger (anchor):** the element the tooltip describes — an icon button, a truncated label, a form field, a badge. The trigger owns the accessibility relationship (`aria-describedby`), not the bubble.
* **Bubble (surface):** the floating container. Padding, small radius, high-contrast background, `role="tooltip"`, a stable `id`.
* **Label text:** short. A few words to a sentence. Plain text only — no headings, no lists, no interactive nodes.
* **Arrow / caret (optional):** the little pointer connecting bubble to trigger. Purely decorative (`aria-hidden`), and the part that makes collision math annoying (§9.4).
* **Positioning context:** the anchor point + preferred placement (top/bottom/start/end) + collision fallback logic. In modern builds this is a library (Floating UI) or native CSS anchor positioning.
* **Timing controller:** the open delay, close delay, and the shared "group" delay so moving between adjacent triggers doesn't re-wait every time.
* **Portal / layer:** where the bubble renders in the DOM (usually a portal to `<body>` or the top layer) so `overflow: hidden` and stacking contexts don't clip it.

## 4\. Sizing & Density
Tooltips are deliberately small — bigness signals "this is real content, put it somewhere permanent." A typical scale:

| Token | Value | Notes |
| ---| ---| --- |
| Max width | 200–320px | Cap it. A tooltip wider than ~40 characters per line is really a popover. |
| Padding (Y / X) | 6px / 10px | Snug. It's a label, not a card. |
| Font size | 12–13px | One step below body. Never below 12px for legibility. |
| Line height | 1.35–1.45 | Tight but readable if it wraps to two lines. |
| Radius | 4–8px | Small. Big radius reads as a card/popover. |
| Arrow size | 6–8px | Optional; skip it for cleaner, easier collision handling. |
| Offset (gap from trigger) | 6–10px | Enough that the arrow and the trigger's focus ring don't collide. |
| Elevation | subtle shadow | `0 2px 8px rgba(0,0,0,.18)`. Just enough to lift off the page. |
| z-index / layer | top layer | Use the top layer (Popover API) or a high z portal; never fight stacking contexts by hand. |

Rules that matter:
* **Cap the width and let text wrap** — never let a tooltip stretch the full viewport.
* **Keep it to one or two lines.** Three-plus lines is a signal the content belongs in a toggletip or inline text.
* **The trigger still owns a ≥44px tap/target-consideration** even though the tooltip itself is display-only.

## 5\. States (the part juniors forget)
A tooltip has fewer _visual_ states than a button but more _timing_ states, and the timing is where it feels cheap or polished.

1. **Hidden (rest)** — the default. Not in the DOM, or present but `hidden`/`display:none`. Not just `opacity:0` (that leaves it hoverable and screen-reader-visible — a bug).
2. **Pending-open (delay)** — pointer entered the trigger; a timer (typically 300–700ms) is running before showing. Cancels if the pointer leaves first. This delay is what stops tooltips firing on every accidental mouse pass.
3. **Visible (open)** — shown and positioned. On keyboard focus it should appear with **no delay** (the user deliberately tabbed there).
4. **Pending-close (grace)** — pointer left the trigger; a short close delay (~100–300ms) lets the user move the pointer _into_ the bubble to read/select text without it vanishing (WCAG 1.4.13 "hoverable").
5. **Dismissed** — user pressed **Esc**; the tooltip closes and stays closed until re-triggered (WCAG 1.4.13 "dismissable"). Focus stays on the trigger.
6. **Positioned / repositioned** — after open, if it would overflow the viewport it flips/shifts to a fallback placement. On scroll or resize it repositions.
7. **Group-active** — when several triggers share a group, the first open waits the full delay; moving to a sibling within a short window opens immediately (the "warm" delay). This is what makes toolbars feel responsive instead of laggy.

The states that separate amateur from senior: the **grace/hoverable** window (state 4) and the **Esc-dismiss** (state 5). Skip either and you fail WCAG 1.4.13. Most tooltip libraries that _look_ done are missing one of them.

## 6\. Types & Variants
By purpose, not just by color:

* **Label tooltip:** the canonical case — names an icon-only control ("Delete", "Bold", "Share"). The button still needs its own `aria-label`; the tooltip is the _visual_ echo for sighted mouse/keyboard users.
* **Description tooltip:** adds a short hint to an already-labelled control ("Applies to all future invoices"). This is the `aria-describedby` case.
* **Truncation tooltip:** reveals full text when a label is ellipsis-clipped. Only show it when the text is _actually_ truncated (§12) — a tooltip that repeats fully-visible text is noise.
* **Toggletip (variant, not a tooltip):** click-to-reveal aside with `aria-live`. Covered in depth in §9.7 because it's the thing people reach for a tooltip to do and shouldn't.
* **Rich tooltip (careful):** Material's term for a tooltip with a little more text and _optionally_ an action. The moment it has an action it's a popover — treat it as one. "Rich tooltip" is a design-system label, not a license to put buttons in a `role="tooltip"`.
* **Error/validation "tooltip":** a common anti-pattern. Validation messages must be persistent and programmatically associated (`aria-describedby` on the input, always in the DOM), not hover-revealed. If it's an error, it's not a tooltip.

## 7\. When to Use One (and When Not To)
Use a tooltip when **all** of these are true: the info is _supplemental_, it's _short_, the trigger _already has an accessible name_, and losing the tooltip entirely wouldn't break the task.

**Perfect fit:**
* Naming an icon-only button in a toolbar.
* A one-line "what does this do" hint on a settings toggle.
* Revealing full text for a truncated cell in a dense table.
* A keyboard-shortcut hint next to a menu item ("Save ⌘S").

**Do NOT use a tooltip when:**
* **The content is essential.** Requirements, instructions, pricing terms, error messages. Put it inline or in a persistent helper. _If the user must know it, don't hide it behind a hover._
* **The content is interactive.** Any link/button/input → popover or dialog.
* **You're on touch-primary UI.** Tooltips barely exist on touch (§9.5). If the info matters on mobile, it can't live only in a tooltip.
* **The trigger has no accessible name and you're using the tooltip** **_as_** **the name.** Fix the name (`aria-label`); don't lean on the tooltip.
* **It's long.** More than ~2 lines → toggletip, inline text, or a "learn more" disclosure.

The senior heuristic: _a tooltip is a garnish, never the meal._ If you catch yourself putting anything load-bearing in one, stop and pick a different component.

## 8\. Tooltips Across Design Systems
Same component, very different philosophies and — importantly — different levels of accessibility rigor. Knowing these makes you fluent in any codebase and warns you which defaults to distrust.

* **Material Design (Google):** Plain tooltip (short label, dark surface) vs. **Rich tooltip** (more text, persists, may have actions — really a popover). Material 3 tooltips appear on hover and long-press (touch), with a built-in show/hide delay. Reasonable a11y defaults, but the "rich" variant tempts misuse.
* **Apple HIG (iOS/macOS):** Historically macOS "help tags" on hover; iOS deliberately has _almost no tooltip_ because touch has no hover. Apple's guidance is essentially "don't hide anything important in a tooltip" — which is the correct instinct baked into a platform.
* **Fluent (Microsoft):** Tooltip is well-specified with `role="tooltip"`, delay, and keyboard support. Fluent also distinguishes tooltip from **TeachingBubble/Callout** (interactive → popover). One of the more correct enterprise defaults.
* **Ant Design / enterprise:** `Tooltip` plus a separate `Popover` (interactive) plus `Popconfirm`. The split is explicit, which is healthy — but Ant's default tooltip historically needed help to be fully keyboard/Esc correct. Verify the version.
* **Carbon (IBM):** Cleanly separates **Tooltip** (definition/label, non-interactive) from **Toggletip** (interactive trigger, popover content). Carbon's docs are among the best public explanations of the tooltip/toggletip line — worth reading.
* **Tailwind / utility-first:** No component; you compose it or reach for a headless lib. Freedom + the risk that everyone rebuilds a broken one. Wrap a real headless primitive.
* **shadcn/Radix (2024+ web standard):** `Tooltip` built on Radix primitives — keyboard, focus, Esc, delay-group, collision all handled. This is the sane default for React (§9.8). Radix draws the tooltip/popover line correctly: `Tooltip` is non-interactive, `Popover`/`HoverCard` are the interactive ones.

The through-line: the _good_ systems all draw the same border — tooltip = non-interactive describe-on-hover-or-focus; anything interactive is a different component. When a system blurs that line (a "rich tooltip" with a button), that's where bugs breed.

## 9\. The Code
The biggest section. A tooltip is mostly _behavior and timing_, so the code carries most of the weight. We build up from the honest native story to production React, then cover the traps.

### 9.1 HTML (the association is the whole game)
There is no `<tooltip>` element. The markup that matters is the **relationship** between trigger and bubble.

```html
<!-- Label tooltip on an icon-only button.
     The button ALREADY has its name via aria-label.
     The tooltip DESCRIBES; it does not label. -->
<button type="button" aria-label="Delete item" aria-describedby="tt-delete">
  <svg aria-hidden="true"><!-- trash icon --></svg>
</button>
<div role="tooltip" id="tt-delete" hidden>Delete item</div>

<!-- Description tooltip on an already-labelled control -->
<label for="apr">APR</label>
<input id="apr" aria-describedby="tt-apr">
<div role="tooltip" id="tt-apr" hidden>Annual rate including fees. Updated daily.</div>
```

Key HTML rules:
* **`role="tooltip"`** **on the bubble;** **`aria-describedby`** **on the trigger, pointing at the bubble's** **`id`****\*\*\*\*.** That pairing is the contract. Screen readers read the trigger's name, then its description.
* **Use** **`hidden`** **(or** **`display:none`****\*\*\*\*), not** **`opacity:0`****\*\*\*\*, for the resting state.** An `opacity:0` bubble is still in the accessibility tree and still hoverable — it announces phantom text and creates dead hover zones.
* **`aria-describedby`****\*\*\*\*, never** **`aria-labelledby`****\*\*\*\*, for a tooltip.** Describe, don't label. If the trigger needs a _name_, that's `aria-label` on the trigger itself.
* **The bubble is inert.** No `tabindex`, no focusable children, `aria-hidden` on the decorative arrow.

### 9.2 CSS (states done right — and the hover trap)
The naive CSS-only tooltip is where careers-worth of accessibility bugs come from. Here's the honest version and its ceiling.

```css
.tooltip {
  position: absolute;               /* positioned by JS or CSS anchor */
  max-width: 240px;
  padding: 6px 10px;
  border-radius: 6px;
  background: #0A0A0A;              /* Night */
  color: #E8E6E1;                   /* Platinum */
  font: 500 12px/1.4 system-ui, sans-serif;
  box-shadow: 0 2px 8px rgba(0,0,0,.28);
  /* Rest state: truly gone, not just transparent */
}
.tooltip[hidden] { display: none; }

/* Fade in only when shown, and only if motion is allowed */
.tooltip { opacity: 0; transition: opacity .12s ease; }
.tooltip[data-open] { opacity: 1; }

@media (prefers-reduced-motion: reduce) {
  .tooltip { transition: none; }
}
```

The pure-CSS `:hover`/`:focus`\-reveal pattern (no JS) is tempting:

```css
.has-tip:hover .tooltip,
.has-tip:focus-within .tooltip { display: block; }
```

**Say it plainly: CSS-only tooltips cannot be made fully compliant.** They have no honest way to:
* be **dismissable with Esc** (no JS = no keydown handler) — fails WCAG 1.4.13;
* do **collision detection** (flip/shift at the viewport edge) — they get clipped;
* implement a **grace period** so you can move into the bubble (`:hover` drops the instant you leave the trigger);
* add an **open delay** without `transition-delay` hacks that also delay the _hide_ and feel wrong.

Use CSS-only tooltips only for non-essential, decorative hints where failing 1.4.13 is a known, accepted tradeoff — and even then, prefer the Popover API path (§9.6). For anything real, JS (or a library) owns the behavior; CSS owns only the skin.

### 9.3 Vanilla JavaScript (the behavior nobody can skip)
This is the reference implementation of a _correct_ tooltip's behavior: hover **and** focus triggers, open delay, close grace period, and Esc-dismiss. Learn this cold — every library is doing exactly this under the hood.

```javascript
function attachTooltip(trigger, bubble, { openDelay = 400, closeDelay = 150 } = {}) {
  let openTimer, closeTimer;

  const show = () => {
    clearTimeout(closeTimer);
    bubble.hidden = false;
    bubble.dataset.open = "";
    position(trigger, bubble);        // see §9.4 collision logic
  };
  const hide = () => {
    clearTimeout(openTimer);
    delete bubble.dataset.open;
    bubble.hidden = true;
  };

  const scheduleShow = (delay) => {
    clearTimeout(openTimer);
    openTimer = setTimeout(show, delay);
  };
  const scheduleHide = () => {
    clearTimeout(closeTimer);
    closeTimer = setTimeout(hide, closeDelay);   // grace period = hoverable
  };

  // Named handlers so teardown can detach every one of them.
  const onTriggerEnter = () => scheduleShow(openDelay);
  const onFocus = () => scheduleShow(0);
  const onBubbleEnter = () => clearTimeout(closeTimer);
  // Dismissable: Esc closes it, focus stays put (WCAG 1.4.13)
  const onKeydown = (e) => { if (e.key === "Escape" && !bubble.hidden) hide(); };

  // Pointer: delayed open, grace-period close
  trigger.addEventListener("pointerenter", onTriggerEnter);
  trigger.addEventListener("pointerleave", scheduleHide);

  // KEYBOARD FOCUS: appears immediately, no delay. This is non-negotiable.
  trigger.addEventListener("focus", onFocus);
  trigger.addEventListener("blur", hide);

  // Hoverable: let the pointer move INTO the bubble without it closing
  bubble.addEventListener("pointerenter", onBubbleEnter);
  bubble.addEventListener("pointerleave", scheduleHide);

  document.addEventListener("keydown", onKeydown);

  // Cleanup: call on teardown/unmount. Without it the GLOBAL keydown listener
  // (and the pending timers) leak — a tooltip per row would strand N document
  // listeners that fire forever after the triggers are gone.
  return function destroy() {
    clearTimeout(openTimer);
    clearTimeout(closeTimer);
    trigger.removeEventListener("pointerenter", onTriggerEnter);
    trigger.removeEventListener("pointerleave", scheduleHide);
    trigger.removeEventListener("focus", onFocus);
    trigger.removeEventListener("blur", hide);
    bubble.removeEventListener("pointerenter", onBubbleEnter);
    bubble.removeEventListener("pointerleave", scheduleHide);
    document.removeEventListener("keydown", onKeydown);
  };
}
```

Everything senior about a tooltip lives in this function:
* **`focus`****/****`blur`** **triggers, not just pointer.** A tooltip that only shows on hover is invisible to keyboard users — a WCAG 2.1.1 failure. Focus shows it with **zero** delay because tabbing there is deliberate.
* **Open delay on pointer** stops the flicker of tooltips firing as the cursor sweeps across a toolbar.
* **Close delay + bubble** **`pointerenter`** = the hoverable grace period. Without it, users can't reach the tooltip to read or select its text.
* **Esc dismiss** = the dismissable requirement. Cheap to add, and its absence is a hard audit fail.
* **Returns a** **`destroy`** that clears the timers and removes the global `keydown` listener. The `keydown` handler is bound to `document`, so without teardown every `attachTooltip` call leaks one permanent listener — call `destroy` on unmount.

### 9.4 Positioning & Collision (the hard part)
A tooltip near the viewport edge that gets clipped or points at nothing is the visual tell of a hand-rolled build. You need: a **preferred placement**, **flip** (top↔bottom when it would overflow), **shift** (slide along the axis to stay on-screen), and an **arrow** that tracks the trigger.

**Do not write this yourself in 2026.** Two good answers:

**(a) Floating UI** — the de-facto library (successor to Popper). It composes middleware: `offset` (gap), `flip` (choose the side with room), `shift` (keep it in view), `arrow` (place the caret).

```javascript
import { computePosition, offset, flip, shift, arrow } from "@floating-ui/dom";

async function position(trigger, bubble, arrowEl) {
  const { x, y, placement, middlewareData } = await computePosition(trigger, bubble, {
    placement: "top",
    middleware: [offset(8), flip(), shift({ padding: 8 }), arrow({ element: arrowEl })],
  });
  Object.assign(bubble.style, { left: `${x}px`, top: `${y}px` });
  // position the arrow from middlewareData.arrow...
}
```

**(b) Native CSS Anchor Positioning** — the platform is catching up. `anchor-name` + `position-anchor` + `position-try` let the browser do flip/shift with no JS. Baseline is still landing across engines in 2026, so feature-detect and fall back to Floating UI.

```css
.trigger { anchor-name: --trig; }
.tooltip {
  position: absolute;
  position-anchor: --trig;
  bottom: anchor(top);                         /* sit above the trigger */
  justify-self: anchor-center;
  position-try-fallbacks: flip-block, flip-inline;   /* auto flip on overflow */
  margin-bottom: 8px;
}
```

When Anchor Positioning is fully baseline this replaces most tooltip-positioning JS. Until then: Floating UI for reliability, CSS anchor as progressive enhancement.

### 9.5 Touch Devices (tooltips don't exist here — design around it)
This is the section teams skip and users on phones pay for. **There is no hover on touch.** A finger either taps (activates the control) or doesn't touch it at all — there is no "point without pressing." So on touch:

* A hover-only tooltip **never appears.** Any information that lives only there is simply _gone_ for mobile users.
* Some platforms map **long-press** to reveal a tooltip (Material does), but it's undiscoverable — users don't know to hold — so you can't rely on it for anything that matters.
* The correct design responses, in order of preference:
    1. **Don't need the tooltip.** Make the control self-evident (label the icon, add visible helper text). Best fix.
    2. **Promote it to a toggletip** (§9.7): a visible ⓘ button the user taps to reveal the aside. This works identically on touch, mouse, and keyboard — often the right call for _any_ platform.
    3. **Inline the info** on small screens (show the helper text permanently below the field instead of on hover).

The rule: **if information only exists in a hover tooltip, assume every touch and keyboard-only user will never see it.** Design so that's acceptable, or don't use a tooltip.

### 9.6 The `title` attribute & the Popover API
**The** **`title`** **attribute is not a tooltip solution.** It's the browser's native hover text, and it's broken in ways you can't fix:
* **No keyboard access** — `title` doesn't show on focus in most browsers. Instant WCAG 2.1.1 fail.
* **No touch access**, no styling, an unpredictable ~1–2s delay you can't control, and it vanishes after a few seconds whether or not you've finished reading.
* **Screen-reader support is inconsistent** — sometimes read, sometimes not, sometimes duplicated with the accessible name.

Use `title` only as a last-ditch, low-stakes fallback (e.g. an abbreviation's expansion where nothing depends on it). Never for anything a user _needs_. If you want the accessible name on an icon button, use `aria-label`, not `title`.

**The Popover API +** **`popover="hint"`** is the emerging native path (§11). The Popover API gives you top-layer rendering, light-dismiss, and declarative wiring. In 2026 a `popover="hint"` value is landing specifically for tooltip-like, non-interactive hints that don't steal focus and auto-dismiss when another hint opens:

```html
<button popovertarget="tip" popovertargetaction="show">Status</button>
<div id="tip" popover="hint" role="tooltip">Synced 2 minutes ago</div>
```

`popover="hint"` is the honest native tooltip we never had — top layer (no clipping), no focus theft, and it coexists with `popover="auto"` menus without fighting them. Feature-detect it; where it's unsupported, fall back to the JS/library path. This is where tooltips are going; wire your abstraction so you can swap to it.

### 9.7 The Toggletip (click-to-reveal, `aria-live`)
The thing people _reach for a tooltip to do_ and shouldn't: an aside the user opens on demand. It's a **button** that reveals content into a **live region**. Different trigger (click, not hover), different ARIA (`aria-live`, not `aria-describedby`), different keyboard model.

```html
<span class="toggletip">
  <button type="button" aria-expanded="false" aria-label="What is a routing number?">
    <svg aria-hidden="true"><!-- info icon --></svg>
  </button>
  <!-- Live region: injecting text here makes SRs announce it -->
  <span role="status" class="toggletip__bubble" hidden></span>
</span>
```

```javascript
const btn = el.querySelector("button");
const bubble = el.querySelector(".toggletip__bubble");
const MSG = "The 9-digit number on the bottom-left of a check.";

btn.addEventListener("click", () => {
  const open = bubble.hidden === false;
  if (open) { bubble.hidden = true; bubble.textContent = ""; btn.setAttribute("aria-expanded", "false"); }
  else {
    bubble.hidden = false;
    // Re-inserting the text is what triggers the aria-live announcement
    bubble.textContent = MSG;
    btn.setAttribute("aria-expanded", "true");
  }
});

document.addEventListener("keydown", (e) => {
  if (e.key === "Escape" && !bubble.hidden) { bubble.hidden = true; bubble.textContent = ""; btn.setAttribute("aria-expanded", "false"); btn.focus(); }
});
// Light-dismiss on outside click, too.
```

Why the toggletip is often the _better_ choice than a tooltip:
* **It works on touch, mouse, and keyboard identically** — a tap/click/Enter, no hover required.
* **The content is intentionally revealed**, so slightly longer text is fine.
* **`role="status"`****/****`aria-live="polite"`** announces the content when it's injected, so screen-reader users get it on demand.

Keep the content **non-interactive** still — the moment you need a link inside, it's a popover.

### 9.8 React + TypeScript (a proper reusable component)
The production pattern: a single `Tooltip` that wires focus + hover triggers, delay, portal, Esc, and the `aria-describedby` relationship, with an SSR-safe generated id. In real apps you'd delegate positioning to Floating UI's React bindings; here's the shape.

```typescript
import {
  cloneElement, isValidElement, useEffect, useId, useRef, useState,
  type ReactElement, type ReactNode,
} from "react";
import { createPortal } from "react-dom";

interface TooltipProps {
  label: ReactNode;              // SHORT, non-interactive text
  children: ReactElement;        // exactly one focusable trigger
  openDelay?: number;
  closeDelay?: number;
  placement?: "top" | "bottom" | "start" | "end";
}

export function Tooltip({
  label, children, openDelay = 400, closeDelay = 150, placement = "top",
}: TooltipProps) {
  const id = useId();
  const [open, setOpen] = useState(false);
  const openT = useRef<number>();
  const closeT = useRef<number>();

  const show = (delay: number) => {
    window.clearTimeout(closeT.current);
    openT.current = window.setTimeout(() => setOpen(true), delay);
  };
  const hide = () => {
    window.clearTimeout(openT.current);
    closeT.current = window.setTimeout(() => setOpen(false), closeDelay);
  };

  useEffect(() => {
    if (!open) return;
    const onKey = (e: KeyboardEvent) => { if (e.key === "Escape") setOpen(false); };
    document.addEventListener("keydown", onKey);
    return () => document.removeEventListener("keydown", onKey);
  }, [open]);

  // Clear any pending open/close timer on unmount, so a tooltip that unmounts
  // mid-delay (e.g. a row that scrolls away) can't fire setOpen after teardown
  // — the classic "state update on an unmounted component" leak.
  useEffect(() => () => {
    window.clearTimeout(openT.current);
    window.clearTimeout(closeT.current);
  }, []);

  if (!isValidElement(children)) {
    throw new Error("Tooltip expects a single element child as the trigger.");
  }

  // Merge our handlers + the describedby onto the REAL trigger element.
  const trigger = cloneElement(children, {
    "aria-describedby": open ? id : undefined,
    onPointerEnter: () => show(openDelay),
    onPointerLeave: hide,
    onFocus: () => show(0),          // focus = instant, no delay
    onBlur: () => setOpen(false),
  });

  return (
    <>
      {trigger}
      {open &&
        createPortal(
          <div
            role="tooltip"
            id={id}
            data-placement={placement}
            className="tooltip"
            onPointerEnter={() => window.clearTimeout(closeT.current)}  // hoverable
            onPointerLeave={hide}
          >
            {label}
          </div>,
          document.body,
        )}
    </>
  );
}
```

Why this is the senior version, not the junior one:
* **`aria-describedby`** **is applied to the child trigger via** **`cloneElement`**, not to a wrapper `<span>`. The relationship must sit on the real control, and the tooltip must never be the child's _label_.
* **`onFocus`** **opens with delay 0**, `onPointerEnter` with the configured delay — the keyboard user isn't punished with a wait.
* **The bubble's own** **`onPointerEnter`** **cancels the close timer** — the hoverable grace period, in React.
* **Esc closes it** via a document listener scoped to the open state.
* **Timers are cleared on unmount** via a mount-scoped cleanup effect, so a tooltip that unmounts mid-delay can't call `setOpen` after teardown.
* **Portaled to** **`document.body`** so `overflow:hidden` ancestors can't clip it.
* **`useId`** gives a stable, SSR-safe id for the association (no `Math.random` mismatch between server and client).

The one thing this sketch delegates in production: real collision/flip. Wire Floating UI's `useFloating` + middleware (§9.4) into `data-placement`/inline styles instead of the static placement.

### 9.9 shadcn/ui + Radix (the 2024+ standard)
For React, don't hand-roll the above unless you're learning. `shadcn/ui`'s Tooltip wraps Radix primitives, which already implement focus + hover, delay-group, Esc, collision, and the correct `role`/`aria-describedby`.

```tsx
import { Tooltip, TooltipContent, TooltipProvider, TooltipTrigger } from "@/components/ui/tooltip";

// Provider sets a SHARED delay so moving between adjacent triggers feels instant
<TooltipProvider delayDuration={400} skipDelayDuration={300}>
  <Tooltip>
    <TooltipTrigger asChild>
      <button aria-label="Delete item"><TrashIcon aria-hidden /></button>
    </TooltipTrigger>
    <TooltipContent side="top" sideOffset={8}>
      Delete item
    </TooltipContent>
  </Tooltip>
</TooltipProvider>
```

What Radix gets right that hand-rolls miss:
* **`asChild`** merges the trigger onto your real `<button>`, so the accessible name stays on the control and the tooltip only _describes_.
* **`TooltipProvider`** **delay-group:** first open waits `delayDuration`; subsequent triggers within `skipDelayDuration` open instantly — the "warm" toolbar feel from §5.
* **Collision-aware** (`collisionPadding`, auto side-flip), **Esc + pointer-down-outside dismiss**, and it renders in a portal.
* Radix deliberately makes `TooltipContent` **non-interactive** and steers you to `Popover`/`HoverCard` when you need interactivity — the component enforces the §1 boundary for you.

One caveat: Radix Tooltip triggers on focus, but for _pointer_ it (correctly) does **not** open on touch — reinforcing §9.5. Don't "fix" that; design around touch instead.

### 9.10 Tailwind CSS (utility-first skin)
Tailwind has no tooltip _behavior_ — it's a skin. Compose the surface with utilities and let JS/Radix drive the logic. The pure-CSS `group-hover` version below is fine only for decorative, non-essential hints (it inherits every §9.2 limitation — no Esc, no focus-only-instant, no collision):

```html
<span class="relative inline-flex group">
  <button aria-label="Info" class="p-2">…</button>
  <span role="tooltip"
        class="pointer-events-none absolute bottom-full left-1/2 mb-2 -translate-x-1/2
               whitespace-nowrap rounded-md bg-neutral-900 px-2.5 py-1.5
               text-xs text-neutral-100 shadow-lg opacity-0
               transition-opacity group-hover:opacity-100 group-focus-within:opacity-100
               motion-reduce:transition-none">
    More info
  </span>
</span>
```

Honest note: `group-focus-within` gets you keyboard reveal, but you still have no Esc-dismiss and no collision handling. For anything real, pair Tailwind's classes with Radix (§9.9) rather than shipping this alone.

### 9.11 Vue, Svelte, Angular
Same semantics, different syntax. The `role="tooltip"` + `aria-describedby` + focus/hover/Esc logic never changes.

**Vue 3** — use a headless lib (Floating Vue / Radix Vue) in production; the shape:

### 9.12 Bootstrap
Bootstrap ships a tooltip component, but with a sharp catch: **it's opt-in and uses** **`title`****\*\*\*\*-driven markup that you must initialize in JS**, and by default it's applied to elements in a way that's easy to get wrong for a11y.

```html
<button type="button" class="btn btn-secondary"
        data-bs-toggle="tooltip" data-bs-placement="top"
        aria-label="Delete item" title="Delete item">
  <i class="bi bi-trash" aria-hidden="true"></i>
</button>
```

```javascript
// Tooltips are NOT initialized automatically — you must opt in:
const triggerList = document.querySelectorAll('[data-bs-toggle="tooltip"]');
[...triggerList].map((el) => new bootstrap.Tooltip(el, { delay: { show: 400, hide: 150 } }));
```

Bootstrap wins on speed but note: it copies `title` into a generated bubble (so keep a real `aria-label` on the control for the name), it triggers on hover+focus by default (good), but you should verify Esc-dismiss and set `trigger: 'hover focus'` explicitly. For icon-only controls, the accessible name must come from `aria-label`, not the Bootstrap `title`.

### 9.13 Web Component (framework-agnostic)
A tooltip that works anywhere, using the platform. The shadow DOM holds only the `<slot>`; the bubble is rendered in the **light DOM** so its `id` resolves from the slotted trigger's `aria-describedby` (ARIA IDREFs don't cross shadow boundaries — see the caveat).

```javascript
class UITooltip extends HTMLElement {
  #timer;
  connectedCallback() {
    this.attachShadow({ mode: "open" });
    const label = this.getAttribute("label") ?? "";
    // Shadow DOM: just the slot. No label interpolation here.
    this.shadowRoot.innerHTML = `<slot></slot>`;
    const trigger = this.querySelector("[slot], button, a, [tabindex]") ?? this.firstElementChild;

    // Build the bubble as a node and set text via textContent — NEVER innerHTML.
    // Interpolating the `label` attribute into innerHTML would be a stored-XSS
    // sink (e.g. label='<img src=x onerror=...>'); see §15.
    const bubble = document.createElement("div");
    const bubbleId = `tt-${Math.random().toString(36).slice(2, 9)}`;
    bubble.id = bubbleId;
    bubble.setAttribute("part", "bubble");
    bubble.setAttribute("role", "tooltip");
    bubble.hidden = true;
    bubble.style.cssText =
      "position:absolute;max-width:240px;padding:6px 10px;border-radius:6px;" +
      "background:#0A0A0A;color:#E8E6E1;font:500 12px/1.4 system-ui;box-shadow:0 2px 8px rgba(0,0,0,.28);";
    bubble.textContent = label;
    // Append to the LIGHT DOM (same tree as the trigger) so aria-describedby
    // can actually resolve the id — a shadow-DOM id could not.
    this.append(bubble);

    const show = (d) => { clearTimeout(this.#timer); this.#timer = setTimeout(() => { bubble.hidden = false; trigger?.setAttribute("aria-describedby", bubbleId); }, d); };
    const hide = () => { clearTimeout(this.#timer); bubble.hidden = true; trigger?.removeAttribute("aria-describedby"); };
    trigger?.addEventListener("pointerenter", () => show(400));
    trigger?.addEventListener("pointerleave", hide);
    trigger?.addEventListener("focus", () => show(0));
    trigger?.addEventListener("blur", hide);
    this.addEventListener("keydown", (e) => e.key === "Escape" && hide());
  }
}
customElements.define("ui-tooltip", UITooltip);
// <ui-tooltip label="Delete item"><button aria-label="Delete item">🗑</button></ui-tooltip>
```

Caveat worth naming: cross-shadow-DOM ARIA associations are genuinely awkward today — the `id` in `aria-describedby` must resolve in the **same tree** as the trigger, and a shadow-DOM id will not. That's exactly why the code above renders the bubble in the light DOM instead of the shadow root. The **Reference Target / ARIA element reflection** proposals are landing to fix this so the bubble can live in the shadow DOM cleanly; until then, keep the trigger and the bubble id in the same DOM tree.

### 9.14 Native Mobile (SwiftUI, Jetpack Compose, Flutter)
Off the web, the "hover tooltip" concept mostly dissolves — because touch has no hover (§9.5). Platforms expose it differently, and mostly for pointer/accessibility, not as primary UI.

**SwiftUI (iOS/macOS):** `.help` is the tooltip — it shows on macOS hover and, crucially, **feeds VoiceOver** on iOS even though there's no visual hover bubble. That dual role is the point.

```swift
    .help("Delete item")            // macOS tooltip + accessibility hint
    .accessibilityLabel("Delete item")
```

**Jetpack Compose (Android):** `TooltipBox` with a `PlainTooltip` (label) or `RichTooltip`, triggered by long-press on touch and hover on pointer devices.

```kotlin

```

**Flutter:** the `Tooltip` widget — shows on long-press (touch) and hover (pointer), and is read by screen readers via its `message`.

```dart

)
```

The through-line: on native, the tooltip's _visual_ role is minor (long-press, discoverable to almost no one), but the same API doubles as the **accessibility description**. So set it for the a11y value, not because touch users will see a bubble — they won't.

### 9.15 Testing (unit + a11y)
A tooltip is behavior and association, so test the behavior and the wiring — never the pixels.

**Component (Vitest + React Testing Library):**

```typescript
import { render, screen } from "@testing-library/react";
import userEvent from "@testing-library/user-event";
import { describe, it, expect } from "vitest";
import { Tooltip } from "./Tooltip";

describe("Tooltip", () => {
  it("appears on keyboard focus (not just hover)", async () => {
    render(<Tooltip label="Delete item"><button aria-label="Delete item">🗑</button></Tooltip>);
    await userEvent.tab();                                  // focus the trigger
    expect(await screen.findByRole("tooltip")).toHaveTextContent("Delete item");
  });

  it("associates via aria-describedby, not as the label", async () => {
    render(<Tooltip label="Delete item"><button aria-label="Delete item">🗑</button></Tooltip>);
    const btn = screen.getByRole("button", { name: "Delete item" });   // name comes from aria-label
    await userEvent.tab();
    const tip = await screen.findByRole("tooltip");
    expect(btn).toHaveAttribute("aria-describedby", tip.id);            // describes, not labels
  });

  it("is dismissable with Escape while focus stays put", async () => {
    render(<Tooltip label="Delete item"><button aria-label="Delete item">🗑</button></Tooltip>);
    const btn = screen.getByRole("button");
    await userEvent.tab();
    expect(await screen.findByRole("tooltip")).toBeInTheDocument();
    await userEvent.keyboard("{Escape}");
    expect(screen.queryByRole("tooltip")).not.toBeInTheDocument();
    expect(btn).toHaveFocus();                                          // focus never moved
  });

  it("contains no interactive elements", async () => {
    render(<Tooltip label="Delete item"><button aria-label="Delete">🗑</button></Tooltip>);
    await userEvent.tab();
    const tip = await screen.findByRole("tooltip");
    expect(tip.querySelector("a, button, input, [tabindex]")).toBeNull();
  });
});
```

**Automated a11y (jest-axe / Playwright + axe):**

```typescript
import { axe } from "jest-axe";
it("has no axe violations when open", async () => {
  const { container } = render(<Tooltip label="Delete item"><button aria-label="Delete">🗑</button></Tooltip>);
  await userEvent.tab();
  expect(await axe(container)).toHaveNoViolations();
});
```

What to actually assert (and skip):
* **Do test:** it opens on **focus** (not only hover), the association is `aria-describedby` → the tooltip's `id`, the trigger's name is unchanged by the tooltip, **Esc** dismisses it, focus stays on the trigger, and the bubble holds **no** interactive nodes.
* **Don't test:** exact pixel coordinates, arrow position, hex colors, class names. Position is Floating UI's job; testing it is brittle and catches nothing real.
* **Note:** axe cannot detect "opens on hover only" or "content is essential" — those are manual keyboard-and-judgment checks. Automated tools catch _maybe_ half of tooltip bugs. Keyboard-test by hand.

## 10\. Accessibility (non-negotiable)
The tooltip has no safe native fallback, so _every_ correctness property is something you actively build. This is the section that separates a passing audit from a lawsuit.

**Semantics & association:**
* Bubble gets `role="tooltip"` and a stable `id`. Trigger gets `aria-describedby="<that id>"`.
* **Describe, don't label.** `aria-describedby`, never `aria-labelledby`. The trigger's _name_ comes from its own text or `aria-label`; the tooltip is the description on top.
* Icon-only triggers **must** have their own accessible name (`aria-label`) independent of the tooltip. The tooltip is not a substitute for a name.

**Keyboard (WCAG 2.1.1):**
* The tooltip **must appear on keyboard** **`focus`**, not hover alone. Hover-only tooltips are invisible to keyboard and screen-reader users — a hard failure.
* Focus shows it with **no delay**; the trigger must itself be focusable (a real `<button>`/`<a>`/form control).

**Content on Hover or Focus (WCAG 1.4.13) — the three that people fail:**
* **Dismissable:** **Esc** hides the tooltip without moving focus. (Also: it must not obscure other content it's covering.)
* **Hoverable:** the pointer can move _off the trigger and onto the tooltip_ without it disappearing — the grace-period close delay (§9.3). Users need to reach it to read/select text.
* **Persistent:** it stays visible until dismissed, focus moves, or it's no longer valid — it does not auto-vanish on a timer while the trigger is still focused/hovered.

**Content rules:**
* **No essential information.** Anything the user _must_ have cannot live only in a tooltip (touch + keyboard + SR users may never get it). Put essentials inline.
* **No interactive content.** No links, buttons, inputs. If it's interactive, it's a popover — full stop (WCAG 4.1.2 and the reachability of that content).

**Contrast & motion:**
* Text vs. bubble background ≥ **4.5:1** (WCAG AA). Dark tooltips on dark UIs are a common silent fail — check it.
* Honor `prefers-reduced-motion`: no scale/slide-in for those users; fade or nothing.

**Screen-reader behavior:** with `aria-describedby`, the SR reads the trigger's name and then its description when focused. For the **toggletip** pattern it's different — the content is announced through an `aria-live`/`role="status"` region because there's no persistent describedby relationship.

## 11\. Innovative & Emerging Ideas (2026-aware)
The tooltip looks solved but the platform is finally giving it real primitives:

* **`popover="hint"`** **(native tooltips, at last):** the Popover API is gaining a `hint` type specifically for non-interactive, non-focus-stealing hints that render in the **top layer** (no clipping, no z-index wars) and auto-dismiss when another hint opens. This is the biggest tooltip change in a decade — it moves tooltip _behavior_ into the browser. Feature-detect and adopt.
* **CSS Anchor Positioning going baseline:** `anchor`, `position-anchor`, and `position-try-fallbacks` do flip/shift natively (§9.4), retiring most positioning JS. Watch the interop dashboard; progressively enhance.
* **Reference Target / ARIA element reflection:** proposals to let ARIA relationships (`aria-describedby`) cross shadow-DOM boundaries cleanly — fixing the web-component tooltip pain in §9.13.
* **Interest Invokers (****`interesttarget`****):** a proposed declarative way to wire "show this on hover/focus/long-press" without JS event plumbing — the tooltip trigger becomes an HTML attribute. Early, but aimed squarely at this component.
* **AI-generated hint text:** contextual tooltips whose copy is generated per user/field. Powerful and dangerous — generated hints must still respect "no essential info, no interactive content," and must be reviewed for accuracy before shipping.

**For UJG:** design the tooltip abstraction now so its _internals_ can swap to `popover="hint"` + CSS anchor positioning without changing the call sites. Keep the public API (`<Tooltip label>`) stable; let the engine modernize underneath.

## 12\. Conversion & UX Killers (the silent mistakes)
None of these throw an error. They quietly erode trust, hide information, and lock out whole user groups.

* **Essential info trapped in a tooltip.** The cardinal sin. Requirements, pricing caveats, error explanations behind a hover = invisible to mobile, keyboard, and SR users. It reads as "works on my machine" and fails everyone else.
* **Hover-only, no focus trigger.** Silently excludes every keyboard and screen-reader user. No error, no log — just an inaccessible product.
* **No open delay.** Tooltips firing the instant the cursor grazes a control turn a toolbar into a flickering mess. Feels broken. 300–700ms delay fixes it.
* **No grace period (not hoverable).** The tooltip vanishes the moment the user moves toward it, so they can't read or copy the text. Enraging, and a 1.4.13 fail.
* **No Esc-dismiss / it obscures content.** A tooltip covering the thing you're trying to read, with no way to dismiss, is a trap. Esc must work.
* **Clipped at the edge.** No collision handling → the tooltip gets cut off by the viewport or an `overflow:hidden` ancestor, pointing at nothing. Portal it and add flip/shift.
* **Tooltip on touch as the only path.** Building a hover tooltip and assuming mobile users will "figure out" a long-press. They won't. The info is simply gone.
* **Redundant truncation tooltips.** Showing a tooltip that repeats text that's already fully visible = noise that trains users to ignore tooltips. Only show on _actual_ truncation.
* **Using** **`title`** **and calling it done.** Inherits every §9.6 flaw. Looks fine to the developer on desktop; fails keyboard, touch, and half of screen readers.

**The through-line:** every one of these fails _quietly_ for the people you're least likely to be testing as. Audit tooltips with the keyboard and a phone, not just your mouse.

## 13\. Advanced Patterns
Senior-move versions of the patterns above.

**Delay grouping (the "warm" toolbar).** Share one timer across a group of triggers: the first tooltip waits the full delay, but once one is open, moving to adjacent triggers opens instantly (within a short window). This is what makes a toolbar feel responsive instead of laggy. Radix's `TooltipProvider` (§9.9) does this; hand-rolled, you keep a module-level "recently open" timestamp.

**TS-enforced non-interactivity.** Encode the §1 rule in the type system so a teammate _can't_ put a button in a tooltip:

```typescript
// Label must be text/number/element-array of inline text — never an interactive node.
type TooltipContent = string | number | ReadonlyArray<string | number>;
interface TooltipProps { label: TooltipContent; children: React.ReactElement; }
// <Tooltip label={<button/>}> now fails to typecheck.
```

**Truncation-aware tooltip.** Only attach the tooltip when the text is actually clipped:

```typescript
const isTruncated = el.scrollWidth > el.clientWidth;   // measure before deciding to show
```

**Token-driven skin.** Same three-tier token approach as the Button doc — primitive → semantic → component — so the tooltip re-themes and dark-modes without touching its logic:

```css
:root {
  --tt-bg: var(--color-surface-inverse, #0A0A0A);
  --tt-fg: var(--color-text-inverse, #E8E6E1);
  --tt-radius: 6px; --tt-shadow: 0 2px 8px rgba(0,0,0,.28);
}
.tooltip { background: var(--tt-bg); color: var(--tt-fg); border-radius: var(--tt-radius); box-shadow: var(--tt-shadow); }
```

**Portal + top layer.** Always render the bubble outside clipping ancestors. Native path: Popover API top layer. Library path: a portal to `document.body` with Floating UI keeping it pinned to the trigger through scroll.

## 14\. Performance & Bundle Cost
A tooltip feels free. At scale it isn't — and the mistakes are specific.

* **Don't mount a tooltip instance per row.** A 500-row table with a `<Tooltip>` wrapping every cell = 500 timers, listeners, and (worse) 500 portals. Use **one delegated tooltip controller** for the table: listen on the container, read `data-tooltip` off the hovered/focused cell, and render a single shared bubble. One instance, N triggers.
* **Lazy-render the bubble.** Keep it out of the DOM until first open (`{open && ...}`), not hidden-but-present ×500. Hidden nodes still cost layout and memory.
* **Position on demand, not on every scroll frame.** Recompute placement with `requestAnimationFrame`\-throttled listeners, or let Floating UI's `autoUpdate` manage it — and tear it down on close so you're not tracking scroll for a closed tooltip.
* **Tree-shake the positioning lib.** Import only the middleware you use from `@floating-ui/dom` (`offset`, `flip`, `shift`), not the whole kitchen sink — same named-import discipline as the Button doc's icon trap.
* **Prefer the native path where available.** `popover="hint"` + CSS anchor positioning is _zero JS_ for behavior and position — the cheapest possible tooltip once support is there.
* **Compositor-only animation.** Animate `opacity` (and `transform` if you must), never `width`/`top`/`height`, so showing a tooltip doesn't trigger layout on the page beneath it.

## 15\. Security
A display-only hover/focus label has almost no attack surface of its own — but it inherits one real risk the moment its text isn't hardcoded. If the bubble body is user- or CMS-authored and you inject it with `innerHTML` (or a framework's raw-HTML escape hatch), you've built a stored-XSS sink; the native `title` attribute is safe here only because the UA renders it as plain text, so any custom bubble must escape or sanitize the string and set it via `textContent`, never `innerHTML`. Second, never treat "hover to reveal" as a security boundary: a tooltip's text ships to every client in the DOM (or the initial payload), so secrets, internal IDs, or authorization-sensitive detail parked there are readable by anyone with devtools — hover is a UI affordance, not access control. Third, tooltips that lazy-fetch their copy from an endpoint inherit that endpoint's authorization and output-escaping obligations: gate the request, and escape the response before it lands in the bubble. Beyond those three, there is genuinely no new surface here — a tooltip rendering static, trusted text through `textContent` is as safe as any other label, and it's more honest to say so than to manufacture threats.

## 16\. Senior-Level Checklist (ship-ready)
Before a tooltip is "done":
- [ ] Content is **supplemental and non-interactive** — nothing essential, no links/buttons/inputs inside.
- [ ] Trigger is a real focusable element with its **own** accessible name (`aria-label` if icon-only).
- [ ] Association is **`aria-describedby`** → bubble `id`; bubble has `role="tooltip"`. (Describe, not label.)
- [ ] Appears on **keyboard focus**, not hover only; focus opens it with **no delay**.
- [ ] **Dismissable:** Esc closes it, focus stays on the trigger.
- [ ] **Hoverable:** a grace-period close delay lets the pointer move onto the bubble.
- [ ] **Persistent:** doesn't auto-vanish on a timer while hovered/focused.
- [ ] Open delay (~300–700ms) on pointer to prevent flicker; shared delay-group for adjacent triggers.
- [ ] **Collision-aware:** flips/shifts at the viewport edge; portaled out of `overflow:hidden`/stacking contexts.
- [ ] Contrast ≥ 4.5:1; honors `prefers-reduced-motion`.
- [ ] **Touch plan exists:** the info isn't _only_ in the tooltip; on touch it's inlined or promoted to a toggletip.
- [ ] Not relying on the `title` attribute for anything that matters.
- [ ] Tested by keyboard and on a phone, not just with a mouse.

## 17\. Visual Styles (the same tooltip, eleven aesthetics)
Design systems are _whose_ rules; visual styles are _which look_. The same semantic tooltip wears any of these skins. **The rule that never changes: the style is skin.** **`role="tooltip"`****\*\*\*\*, the** **`aria-describedby`** **association, focus-triggering, Esc-dismiss, the hoverable grace period, and contrast stay identical across all eleven.** See the full Design Styles (visual languages) guide.

* **Flat:** solid dark surface, small radius, crisp edge, minimal shadow. Reads instantly; the safest default.
* **Material:** slightly elevated, medium-emphasis surface with Material 3's motion (subtle scale-in from the trigger), tuned show/hide delays.
* **Glassmorphism:** translucent bubble + `backdrop-filter: blur`. Pretty over rich backgrounds, but **guard contrast** — text on a blurred, variable backdrop routinely fails 4.5:1. Keep a solid fallback.
* **Liquid Glass (the 2026 trend):** Apple's iOS 26 / macOS Tahoe material — the pill bubble _refracts_ what's behind it with a specular rim and a faint moving sheen, not a static blur. Gorgeous on the dark Eminence UJG surface. **Guard contrast and** **`prefers-reduced-motion`** (kill the sheen). On SwiftUI this is native `.glassEffect`; on web it's an honest approximation.
* **Neumorphism:** same-color surface with dual soft shadows. Contrast-poor by nature — bad for a text bubble that must hit 4.5:1. Accent use only, if at all.
* **Skeuomorphism:** a "real" little label with bevel, inner highlight, and a subtle gradient — the classic OS help-tag look.
* **Neo-Brutalism:** hard offset shadow, thick black border, zero radius, high-contrast fill. Loud personality; ironically the _best_ contrast of the decorative styles.
* **Claymorphism:** puffy, large-radius bubble with inner top-light and soft outer glow. Friendly; watch that the big radius doesn't make it read as a popover.
* **Aurora / Gradient:** animated multi-hue gradient surface on dark. Premium campaign feel; **honor** **`prefers-reduced-motion`** and keep text contrast on the lightest gradient stop.
* **Minimal / Swiss:** near-zero decoration — a thin hairline, tight type, lots of restraint. Typography does the work.
* **UJG Brand:** the house default — Platinum text on a deep Night/Eminence bubble, Goldenrod hairline or arrow accent, confident-but-small radius, a warm low glow. Afro-Futurist and legible. Lean Liquid Glass accents over the Eminence surface for hero/marketing contexts; keep Flat/Swiss for dense data UIs.

---

**Bottom line:** a tooltip is ten lines of markup wrapped around a dozen accessibility obligations. The visual is the easy 10%; the real component is _behavior_ — it must appear on focus, be dismissable, be hoverable, never hold anything essential or interactive, survive the viewport edge, and have an honest answer for touch. Master the tooltip-vs-toggletip-vs-popover line and WCAG 1.4.13, wire your abstraction so it can swap to `popover="hint"` later, and you've handled the component that trips up more real-world builds than almost anything its size.
