      import { useCallback, useEffect, useId, useRef, useState } from "react";

      export interface SplitMenuItem {
        id: string;
        label: string;
        onSelect: () => void;
        disabled?: boolean;
      }

      export interface SplitButtonProps {
        primaryLabel: string;
        onPrimary: () => void;
        items: SplitMenuItem[];
        menuLabel: string;           // accessible name for the toggle
        disabledPrimary?: boolean;
        disabledToggle?: boolean;
        loading?: boolean;
      }

      export function SplitButton({
        primaryLabel, onPrimary, items, menuLabel,
        disabledPrimary = false, disabledToggle = false, loading = false,
      }: SplitButtonProps) {
        const [open, setOpen] = useState(false);
        const [active, setActive] = useState(0);
        const menuId = useId();
        const toggleRef = useRef<HTMLButtonElement>(null);
        const itemRefs = useRef<Array<HTMLButtonElement | null>>([]);

        const close = useCallback((returnFocus = true) => {
          setOpen(false);
          if (returnFocus) toggleRef.current?.focus();
        }, []);

        useEffect(() => {
          if (open) itemRefs.current[active]?.focus();
        }, [open, active]);

        useEffect(() => {
          if (!open) return;
          const onDocDown = (e: MouseEvent) => {
            if (!(e.target as Element).closest(`[data-split="${menuId}"]`)) setOpen(false);
          };
          document.addEventListener("mousedown", onDocDown);
          return () => document.removeEventListener("mousedown", onDocDown);
        }, [open, menuId]);

        const openMenu = (index = 0) => {
          if (disabledToggle) return;
          setActive(index);
          setOpen(true);
        };

        const onMenuKey = (e: React.KeyboardEvent) => {
          const last = items.length - 1;
          switch (e.key) {
            case "ArrowDown": e.preventDefault(); setActive((i) => (i >= last ? 0 : i + 1)); break;
            case "ArrowUp":   e.preventDefault(); setActive((i) => (i <= 0 ? last : i - 1)); break;
            case "Home":      e.preventDefault(); setActive(0); break;
            case "End":       e.preventDefault(); setActive(last); break;
            case "Escape":    e.preventDefault(); close(); break;
            case "Tab":       close(false); break;
          }
        };

        return (
          <div className="split" role="group" aria-label={`${primaryLabel} actions`} data-split={menuId}>
            <button type="button" className="split__primary"
                    onClick={onPrimary} disabled={disabledPrimary || loading} aria-busy={loading}>
              {loading ? <span className="split__spinner" aria-hidden /> : null}
              {primaryLabel}
            </button>
            <span className="split__seam" aria-hidden="true" />
            <button ref={toggleRef} type="button" className="split__toggle"
                    aria-haspopup="menu" aria-expanded={open} aria-controls={menuId}
                    aria-label={menuLabel} disabled={disabledToggle}
                    onClick={() => (open ? close(false) : openMenu(0))}
                    onKeyDown={(e) => {
                      if (e.key === "ArrowDown") { e.preventDefault(); openMenu(0); }
                      if (e.key === "ArrowUp")   { e.preventDefault(); openMenu(items.length - 1); }
                    }}>
              <svg className="split__caret" aria-hidden viewBox="0 0 20 20"><path d="M6 8l4 4 4-4" fill="none" stroke="currentColor" strokeWidth="1.5" /></svg>
            </button>
            {open && (
              <ul className="split__menu" id={menuId} role="menu" aria-label={menuLabel} onKeyDown={onMenuKey}>
                {items.map((it, i) => (
                  <li role="none" key={it.id}>
                    <button ref={(el) => (itemRefs.current[i] = el)} role="menuitem" type="button"
                            tabIndex={i === active ? 0 : -1} disabled={it.disabled}
                            onClick={() => { it.onSelect(); close(); }}>
                      {it.label}
                    </button>
                  </li>
                ))}
              </ul>
            )}
          </div>
        );
      }

## 1\. What It Actually Is

A **split button** is a single compound control made of **two distinct interactive elements** side by side: a **primary action button** (the obvious default — "Save", "Send", "Deploy") and a smaller **menu toggle** (usually a caret ▾) that opens a menu of related, less-common actions. A thin visual seam divides them, but they are **two separate** **`<button>`** **elements** with **two separate accessibility semantics** and **two separate hit targets**. Clicking the primary fires the default action immediately. Clicking the toggle opens a `role="menu"` popup.

The whole point: surface the one action 80% of users want as a one-click target, while keeping the long tail of variants one click deeper — without a second control eating layout space.

**The 2-4 things people confuse it with:**

* **Dropdown Menu button (menu button)** — a single button whose _only_ job is to open a menu. It has no default primary action; clicking anywhere on it opens the menu. A split button ALWAYS has a committed default action on the primary half. If clicking the main body opens a menu instead of doing something, it is a menu button, not a split button.
* **Segmented Control** — a row of 2+ equal-weight options where selecting one changes a mode/filter (like radio buttons styled as a bar). Segmented controls express _mutually exclusive state_; split buttons express _one action + a menu of actions_. No segment is "primary."
* **Button group / toolbar** — several equal-emphasis buttons grouped for visual proximity (Bold / Italic / Underline). Every member is independently important; there is no primary-plus-overflow relationship and no popup menu.
* **Combo button / gesture button** — a single button that does one thing on tap and opens a menu on long-press or right-click. Split buttons make both affordances _visible and separately clickable_; combo buttons hide the second affordance behind a gesture.

**Rule of thumb:** It is a split button when there is **one action a user reaches for far more than the rest**, AND **2–5 closely related variants of that same action** worth surfacing. If the actions are equal in weight → button group or segmented control. If there is no obvious default → menu button. If there is only one action → plain button.

## 2\. Why It Matters

* **Conversion / speed of the primary path.** The split button exists to make the dominant action a single, large, unmissable click while demoting rarer variants. Collapse it into a plain menu button and you add a click + a cognitive scan to the action 80% of users want. In editors, mail clients, and CI/deploy tools, that primary path is fired thousands of times a day; one extra interaction per fire is a real productivity tax.
* **Accessibility (this is the crux and the usual failure).** A split button is **not one button** and must never be coded as one. It is a primary `<button>` plus a toggle `<button aria-haspopup="menu" aria-expanded>` controlling a `role="menu"`. Screen-reader users need the toggle announced as a menu opener distinct from the primary action, a correct keyboard model (Enter/Space fires primary; the toggle opens the menu; ArrowDown moves in; Escape closes and returns focus), and roving tabindex inside the menu. Get the seam right visually but fuse the semantics and you ship a control that is unusable without a mouse.
* **Trust & error prevention.** The primary action and the "dangerous cousin" (e.g. "Delete" vs "Delete and purge history") sit millimeters apart. If the two halves are not clearly distinct targets — each ≥44px, with a visible seam and independent focus rings — users misfire. A misfire on a destructive menu item is a data-loss incident, not a cosmetic bug.
* **Legal / target-size compliance (WCAG 2.2).** Success Criterion 2.5.8 (Target Size Minimum) requires interactive targets to be at least 24×24 CSS px, with 44×44 the AAA / mobile-practical bar. The caret toggle is the classic violation: designers shrink it to 20px and it fails. Two adjacent targets also must not have overlapping hit areas that trigger the wrong action.

## 3\. Anatomy

Interactive component — every named part:

* **Group container** — the wrapper (`role` NOT set to button; it is just a styled `div`/`span` or a `role="group"` with an `aria-label`). Owns the shared outer border-radius and the overall footprint. Must not itself be focusable or clickable.
* **Primary action button** — the left/leading `<button type="button">` (or `type="submit"` inside a form). Carries the default label + optional leading icon. Fires immediately on click/Enter/Space. Has its own accessible name ("Save").
* **Visual seam / divider** — a 1px rule (or subtle inset shadow) between the two halves. Purely decorative; must not be a focusable element and must not swallow clicks. Its job is to signal "two targets, not one."
* **Menu toggle button** — the trailing `<button type="button" aria-haspopup="menu" aria-expanded="false">` with a caret (▾) glyph. Its accessible name must describe the menu, not just be "▾" — e.g. `aria-label="More save options"`. Owns `aria-controls` pointing at the menu, `aria-expanded` reflecting open state.
* **Caret / disclosure glyph** — the ▾ icon inside the toggle, `aria-hidden="true"`. Often rotates 180° on open (respect reduced-motion).
* **Menu (popup)** — `role="menu"` (or `menu`\-flavored listbox), positioned below-end of the group, containing the alternate actions. Anchored to the toggle, not the primary.
* **Menu items** — each `role="menuitem"` (or `menuitemradio`/`menuitemcheckbox` when stateful). Managed with **roving tabindex** (one item `tabindex="0"`, the rest `-1`). Optional leading icons, trailing shortcut hints, separators (`role="separator"`).
* **Focus rings (two, independent)** — the primary and the toggle each render their own visible focus indicator. The seam ensures they read as distinct.

## 4\. Sizes / Scale / Density

Height is shared across both halves (they must align to one baseline). Width differs: the primary flexes to its label; the toggle is a fixed square-ish target. **The toggle is the constraint** — it must stay ≥24px (WCAG 2.5.8 AA) and ideally ≥44px on touch.

| Tier | Token | Shared height | Primary min-width | Toggle width | Font | Icon | Use case |
| ---| ---| ---| ---| ---| ---| ---| --- |
| Small | `--split-sm` | 32px | 64px | 32px | 13px / `--fs-sm` | 16px | Dense toolbars, data-grid row actions, desktop-only IDE chrome |
| Medium (default) | `--split-md` | 40px | 88px | 40px | 14px / `--fs-md` | 18px | App default — forms, editors, dashboards |
| Large | `--split-lg` | 48px | 112px | 48px | 16px / `--fs-lg` | 20px | Touch, marketing/checkout, primary CTAs |

Rules that matter:

* **Target size:** toggle width = shared height keeps it a square ≥ the height tier. At Small (32px) you are below the 44px touch ideal but still clear the 24px AA floor — Small is desktop-pointer only. On touch, never ship below Medium.
* **Seam spacing:** the divider is 1px; do NOT add internal padding that shrinks either half's hit area below the tier height. Hit area = full half including padding.
* **Spacing tokens:** primary horizontal padding `--space-3` (12px) at md, `--space-4` (16px) at lg; toggle padding is symmetric so the caret is optically centered.
* **Responsive behavior:** below ~360px viewport, consider collapsing to a single menu button (primary becomes the first/emphasized menu item) rather than shrinking the toggle into an unhittable sliver.
* **Menu width:** menu min-width = group width so it visually belongs to the control; max-width caps at ~320px with truncation + `title` for long item labels.

## 5\. States

Each half has its own states, plus compound states for the whole control.

**Primary half:**

* **Default** — resting fill/label per variant.
* **Hover** — background darkens ~8% (solid) or fills (outline). Hover applies to the primary half ONLY, not the toggle — they highlight independently so users see two targets.
* **Focus (focus-visible)** — 2px offset ring on the primary alone. Keyboard focus lands here first in tab order.
* **Active / pressed** — momentary depress (translateY 1px or darker fill) while the pointer/space is down.
* **Loading** — primary shows an inline spinner replacing (or preceding) the label; `aria-busy="true"` on the primary; the primary is disabled during the async action. The toggle may stay enabled or also disable — decide per flow (see §6 disabled variants).
* **Disabled** — reduced opacity, no pointer events, `disabled` attribute. Can be disabled independently of the toggle.

**Menu toggle half:**

* **Default (collapsed)** — caret pointing down, `aria-expanded="false"`.
* **Hover** — its own hover highlight, independent of primary.
* **Focus** — its own 2px ring.
* **Open / expanded** — `aria-expanded="true"`, caret rotated 180°, often a persistent pressed background so it reads "this is what's open." Menu is visible and focus-trapped-ish (arrow keys cycle, Escape exits).
* **Disabled** — independently disable-able; a split button can have an enabled primary and a disabled toggle (no alternates available) or vice versa.

**Compound / component-specific states:**

* **Both-disabled** — entire control inert (e.g. no permission to act at all).
* **Primary-loading + toggle-live** — the async save is running but the user may still open "Save options" to cancel/change — deliberate, only when the menu actions are safe mid-flight.
* **Menu-item-active (roving focus)** — the currently focused `menuitem` inside the open menu; visually highlighted, `tabindex="0"`, others `-1`.
* **Error** — the primary action failed; surface via an `aria-live` status region near the control (NOT by silently reverting). The primary returns to default/enabled so the user can retry.

## 6\. Types / Variants

Interactive component — functional variants:

* **Solid split** — filled primary (brand fill) + a toggle in the same fill family, seam as a lightened/darkened 1px rule. The default high-emphasis CTA form.
* **Outline split** — bordered, transparent fill; the seam is a continuation of the border. Medium emphasis, used when the split button sits among other content and shouldn't dominate.
* **Ghost / subtle split** — no border until hover; used in dense toolbars and row-hover action clusters.
* **Icon-leading primary** — primary carries an icon + label ("⤓ Export"); toggle offers "Export as CSV / JSON / PDF." Very common.
* **Icon-only primary + toggle** — primary is a single glyph with an `aria-label`; toggle offers variants. Space-constrained toolbars. Both halves still ≥ target size.
* **Destructive split** — primary is a dangerous default (Delete) in a danger color; menu holds the even-more-dangerous cousins (Delete permanently). Requires extra confirmation on the destructive menu items.
* **Menu-includes-primary vs menu-excludes-primary** — see the decision below; this is a genuine variant choice, not a skin.

**The "primary action is also in the menu?" decision.** Two valid patterns:

1. **Menu excludes the primary** (Windows/Fluent default) — the menu lists only the _alternates_ ("Save as…", "Save all"). Rationale: the primary is already one click away on the big half, so repeating it wastes a row and implies redundancy.
2. **Menu includes the primary as the first, checked/emphasized item** (common on mobile-collapsed and macOS-flavored patterns) — the menu leads with the current default (often marked ✓), then the alternates. Rationale: consistency when the same menu also renders on small screens where the split has collapsed to a single menu button, and it lets users see "what the big button does" explicitly.

Pick per platform and stay consistent across your product. If your split button collapses to a menu button on mobile, use pattern 2 so the primary never disappears.

## 7\. When to Use (and When Not To)

**Use when:**

* There is **one clearly dominant action** plus **2–5 closely related variants**. Save / Save as / Save a copy. Send / Schedule send / Send and archive. Deploy / Deploy to staging / Deploy with options. Merge / Squash and merge / Rebase and merge (GitHub's canonical example).
* The variants are **variations on the same verb**, not unrelated commands. "Save as / Save a copy" belong; "Save / Delete / Share" do not (those are a toolbar).
* You want to **conserve horizontal space** while keeping the primary a one-click target.

**Don't use / use something else when:**

* **All actions are equal weight** → **segmented control** (mode switch) or **button group** (independent equal actions). No action deserves the big half.
* **There is no default action** — every option requires a choice → **menu button** (single button, `aria-haspopup`, no committed primary).
* **There is only one action** → **plain button**. Never ship a split button with an empty or single-item menu.
* **The alternates are a long list (6+), or need search/grouping/icons-heavy layout** → **menu button** or a proper **dropdown/combobox** — a split button's menu should be short.
* **Primary touch target constraints** — on a cramped mobile UI where the toggle can't reach ≥44px without crowding the primary, **collapse to a menu button** and put the primary as the top menu item.

**Placement heuristics:** anchor the menu **below-end** (aligned to the toggle's trailing edge) so it grows toward available space; flip to above-end near the viewport bottom. Keep split buttons in action zones (form footers, toolbar right side, card footers), never inline in body text.

## 8\. Across Design Systems

* **Material Design 3** — no first-class "split button" primitive; the pattern is composed from a filled/outlined button + an adjacent icon button opening a `Menu`. MD3 leans on the **menu button** and **segmented button** instead, so teams building split buttons on Material assemble them and wire ARIA by hand. Watch the ripple: keep it per-half, not across the seam.
* **Apple HIG** — the **pop-up button** and **pull-down button** cover most cases; the split-button idea appears as a push button paired with a menu-triggering chevron in toolbars. HIG stresses the chevron affordance and that the default action is explicit. On iOS, prefer a menu-attached button; the classic desktop split is a macOS/AppKit toolbar idiom.
* **Fluent (Microsoft)** — has an explicit **SplitButton** component. This is the reference implementation: a primary action button + a menu button with `aria-haspopup`/`aria-expanded`, `menu` role, arrow-key navigation, Escape to close. Fluent's default excludes the primary from the menu. Best mental model to code against.
* **Ant Design** — `Dropdown.Button` is a real split button: `<Button>` + a caret `<Button>` with a `menu` prop. Props include `type`, `size`, `loading`, `disabled` (disables both), and `buttonsRender` to customize each half. Menu is an Ant `Menu` with `items`.
* **Tailwind CSS** — utility-only; no component. You compose two `<button>`s in an `inline-flex` with `rounded-l`/`rounded-r`, a `-ml-px` border seam, and `focus-visible:ring`. Headless behavior comes from Headless UI's `Menu` or Radix.
* **shadcn/ui + Radix** — no packaged SplitButton, but the idiomatic build is a `<Button>` beside a `<DropdownMenu>` whose `DropdownMenuTrigger` is a second `<Button>`. Radix handles `aria-haspopup`, `aria-expanded`, roving focus, typeahead, and Escape/return-focus for free. This is the recommended React path.
* **Bootstrap 5** — first-class **split button dropdown**: `.btn-group` with a `.btn` primary + a `.btn.dropdown-toggle.dropdown-toggle-split` and a `.dropdown-menu`. Bootstrap's JS manages `aria-expanded` and outside-click; you add `.visually-hidden` text to the toggle for its accessible name.
* **Ant / Chakra / Mantine** — Chakra and Mantine expose `Menu` + `Button.Group`; Mantine has a documented "SplitButton" recipe in its docs using `Menu` + a bordered button pair.

### 8b. What's trending in 2026

* **Anchor Positioning API** — `position-anchor` / `position-area` in CSS now positions the menu relative to the toggle with no JS popper library, including automatic flip via `position-try-fallbacks`. Kills a whole class of misalignment bugs and shrinks bundles.
* **`popover`** **+** **`command`****/****`commandfor`** **invokers** — the native Invoker Commands proposal lets the toggle declaratively open a `popover` menu with no click handler, improving resilience and a11y defaults.
* **`::backdrop`****\-free light-dismiss menus** — native popover top-layer rendering escapes `overflow:hidden` clipping that historically broke split-button menus inside scroll containers.
* **Liquid Glass skins** (Apple iOS 26 / macOS Tahoe) — split buttons rendered as a single translucent lozenge with a specular seam; the design challenge is keeping the two targets legibly distinct under heavy blur.

```plain
import { useCallback, useEffect, useId, useRef, useState } from "react";

export interface SplitMenuItem {
  id: string;
  label: string;
  onSelect: () => void;
  disabled?: boolean;
}

export interface SplitButtonProps {
  primaryLabel: string;
  onPrimary: () => void;
  items: SplitMenuItem[];
  menuLabel: string;           // accessible name for the toggle
  disabledPrimary?: boolean;
  disabledToggle?: boolean;
  loading?: boolean;
}

export function SplitButton({
  primaryLabel, onPrimary, items, menuLabel,
  disabledPrimary = false, disabledToggle = false, loading = false,
}: SplitButtonProps) {
  const [open, setOpen] = useState(false);
  const [active, setActive] = useState(0);
  const menuId = useId();
  const toggleRef = useRef<HTMLButtonElement>(null);
  const itemRefs = useRef<Array<HTMLButtonElement | null>>([]);

  const close = useCallback((returnFocus = true) => {
    setOpen(false);
    if (returnFocus) toggleRef.current?.focus();
  }, []);

  useEffect(() => {
    if (open) itemRefs.current[active]?.focus();
  }, [open, active]);

  useEffect(() => {
    if (!open) return;
    const onDocDown = (e: MouseEvent) => {
      if (!(e.target as Element).closest(`[data-split="${menuId}"]`)) setOpen(false);
    };
    document.addEventListener("mousedown", onDocDown);
    return () => document.removeEventListener("mousedown", onDocDown);
  }, [open, menuId]);

  const openMenu = (index = 0) => {
    if (disabledToggle) return;
    setActive(index);
    setOpen(true);
  };

  const onMenuKey = (e: React.KeyboardEvent) => {
    const last = items.length - 1;
    switch (e.key) {
      case "ArrowDown": e.preventDefault(); setActive((i) => (i >= last ? 0 : i + 1)); break;
      case "ArrowUp":   e.preventDefault(); setActive((i) => (i <= 0 ? last : i - 1)); break;
      case "Home":      e.preventDefault(); setActive(0); break;
      case "End":       e.preventDefault(); setActive(last); break;
      case "Escape":    e.preventDefault(); close(); break;
      case "Tab":       close(false); break;
    }
  };

  return (
    <div className="split" role="group" aria-label={`${primaryLabel} actions`} data-split={menuId}>
      <button type="button" className="split__primary"
              onClick={onPrimary} disabled={disabledPrimary || loading} aria-busy={loading}>
        {loading ? <span className="split__spinner" aria-hidden /> : null}
        {primaryLabel}
      </button>
      <span className="split__seam" aria-hidden="true" />
      <button ref={toggleRef} type="button" className="split__toggle"
              aria-haspopup="menu" aria-expanded={open} aria-controls={menuId}
              aria-label={menuLabel} disabled={disabledToggle}
              onClick={() => (open ? close(false) : openMenu(0))}
              onKeyDown={(e) => {
                if (e.key === "ArrowDown") { e.preventDefault(); openMenu(0); }
                if (e.key === "ArrowUp")   { e.preventDefault(); openMenu(items.length - 1); }
              }}>
        <svg className="split__caret" aria-hidden viewBox="0 0 20 20"><path d="M6 8l4 4 4-4" fill="none" stroke="currentColor" strokeWidth="1.5" /></svg>
      </button>
      {open && (
        <ul className="split__menu" id={menuId} role="menu" aria-label={menuLabel} onKeyDown={onMenuKey}>
          {items.map((it, i) => (
            <li role="none" key={it.id}>
              <button ref={(el) => (itemRefs.current[i] = el)} role="menuitem" type="button"
                      tabIndex={i === active ? 0 : -1} disabled={it.disabled}
                      onClick={() => { it.onSelect(); close(); }}>
                {it.label}
              </button>
            </li>
          ))}
        </ul>
      )}
    </div>
  );
}
```

```plain
function initSplitButton(root) {
  const toggle = root.querySelector(".split__toggle");
  const menu = root.querySelector(".split__menu");
  const items = () => Array.from(menu.querySelectorAll('[role="menuitem"]:not(:disabled)'));
  let active = -1;

  function open(index = 0) {
    menu.hidden = false;
    toggle.setAttribute("aria-expanded", "true");
    focusItem(index);
    document.addEventListener("mousedown", onOutside, true);
  }
  function close(returnFocus = true) {
    menu.hidden = true;
    toggle.setAttribute("aria-expanded", "false");
    document.removeEventListener("mousedown", onOutside, true);
    if (returnFocus) toggle.focus();
  }
  function focusItem(i) {
    const list = items();
    if (!list.length) return;
    active = (i + list.length) % list.length;
    list.forEach((el, n) => el.setAttribute("tabindex", n === active ? "0" : "-1"));
    list[active].focus();
  }
  function onOutside(e) { if (!root.contains(e.target)) close(false); }

  toggle.addEventListener("click", () =>
    toggle.getAttribute("aria-expanded") === "true" ? close(false) : open(0));
  toggle.addEventListener("keydown", (e) => {
    if (e.key === "ArrowDown") { e.preventDefault(); open(0); }
    if (e.key === "ArrowUp")   { e.preventDefault(); open(items().length - 1); }
  });
  menu.addEventListener("keydown", (e) => {
    switch (e.key) {
      case "ArrowDown": e.preventDefault(); focusItem(active + 1); break;
      case "ArrowUp":   e.preventDefault(); focusItem(active - 1); break;
      case "Home":      e.preventDefault(); focusItem(0); break;
      case "End":       e.preventDefault(); focusItem(items().length - 1); break;
      case "Escape":    e.preventDefault(); close(); break;
      case "Tab":       close(false); break;
    }
  });
  menu.addEventListener("click", (e) => {
    if (e.target.closest('[role="menuitem"]')) close();
  });
}
```

## 10\. Accessibility

Standalone. The split button's #1 failure is being coded as one button — this section is the antidote.

**Roles & ARIA (the correct model):**

* The group wrapper: `role="group"` + `aria-label` (e.g. "Save actions"). It is NOT a button and NOT focusable.
* Primary: a real `<button>` with a visible/text accessible name. No `aria-haspopup` — it does one thing.
* Toggle: a separate `<button>` with `aria-haspopup="menu"`, `aria-expanded="false"→"true"`, `aria-controls="<menu id>"`, and its OWN accessible name describing the menu (`aria-label="More save options"`) — never just the caret glyph.
* Menu: `role="menu"` with `aria-label` matching the toggle's intent. Children are `role="menuitem"` (or `menuitemradio`/`menuitemcheckbox` when stateful), separators `role="separator"`.
* Roving tabindex inside the menu: exactly one item `tabindex="0"`, the rest `tabindex="-1"`. The menu container is not in the tab sequence.

**Keyboard map (interactive — required):**

| Key | Focus on primary | Focus on toggle | Focus inside open menu |
| ---| ---| ---| --- |
| `Tab` | moves to toggle | moves out of control (closes menu if open) | closes menu, moves to next control |
| `Shift+Tab` | moves out (leading) | moves to primary | closes menu, returns to toggle then out |
| `Enter` / `Space` | fires primary action | opens menu (focus first item) | activates focused menu item, closes, returns focus to toggle |
| `ArrowDown` | — | opens menu, focus first item | move to next item (wraps) |
| `ArrowUp` | — | opens menu, focus last item | move to previous item (wraps) |
| `Home` / `End` | — | — | first / last item |
| `Escape` | — | — | closes menu, returns focus to toggle |
| `A–Z` (typeahead) | — | — | jump to next item starting with that letter |

**Focus management rules:**

* On menu open → move focus into the menu (first item for ArrowDown/Enter, last item for ArrowUp).
* On Escape or selection → close and **return focus to the toggle** (never drop focus to `<body>`).
* On outside-click or Tab-out → close WITHOUT stealing focus (focus is already leaving).
* Primary and toggle each get an independent visible focus ring; the seam guarantees they read as two targets.

**Contrast:** text/label ≥ 4.5:1 (WCAG 1.4.3). The seam divider and the caret glyph are non-text UI — ≥ 3:1 against adjacent fills (1.4.11). The focus ring ≥ 3:1 against both the button and the page background. Don't let the seam be the ONLY thing distinguishing the two halves for low-vision users — pair it with the hover/focus behaving per-half.

**Target size (WCAG 2.5.8):** each half ≥ 24×24 CSS px (AA), ≥ 44×44 recommended for touch. The toggle is the usual offender — never shrink the caret target below the shared button height. Adjacent targets must not overlap hit areas.

**`prefers-reduced-motion`****:** disable the caret's rotate transition and the active-press translate; the menu should appear/dismiss without slide/scale animation (instant or a short opacity fade only).

**Common failures specific to split buttons:**

* Coding the whole thing as one `<button>` (or a `<div onclick>`) — screen readers announce one control; keyboard users can't reach the menu.
* Toggle's accessible name is "▾" or empty — announced as "button" with no purpose.
* Missing/incorrect `aria-expanded` (stuck at false) — SR users never learn the menu opened.
* Menu items not using roving tabindex — Tab walks through every item instead of arrows cycling; focus escapes the menu.
* Escape doesn't return focus to the toggle — keyboard user is dumped at the top of the page.
* Focus ring shared across both halves so it's ambiguous which target is active.
* Menu clipped by `overflow:hidden` on an ancestor — visible to mouse users who scroll, invisible/awkward otherwise.

## 11\. Innovative / Emerging Ideas

* **CSS Anchor Positioning for the menu (2026).** Position the `role="menu"` with `position-anchor`/`position-area: bottom span-left` tied to the toggle, and `position-try-fallbacks: flip-block` for automatic above/below flipping — no Popper.js, no ResizeObserver, no JS at all for placement. Escapes clipping via the top layer when combined with `popover`.
* **Invoker Commands (****`command`****/****`commandfor`****).** The toggle becomes `<button commandfor="menu" command="toggle-popover">` and the menu a native `popover`, so open/close, light-dismiss, and `aria-expanded` sync are handled declaratively by the platform — the split button works before your JS hydrates.
* **Smart default via usage telemetry.** The primary half's action adapts to what the user picks most from the menu (respecting a "pin default" escape hatch). The 3rd-most-used variant graduates to the big button; announce the change politely and never move a destructive action into the default slot.
* **View Transitions on default-swap.** When the primary label changes (smart default, or a mode switch), animate the swap with the View Transitions API so the control's identity is preserved visually — gated behind `prefers-reduced-motion`.
* **Speculation Rules on menu hover/focus.** For navigational split buttons (e.g. "Open / Open in new tab / Open preview"), prerender/prefetch the likely target when a menu item is focused, so selection feels instant.

## 12\. Conversion / UX Killers

* **Toggle target too small.** A 20px caret next to a 44px primary fails WCAG 2.5.8 and causes misfires — users hit the primary when reaching for options, or miss entirely on touch. _Fix:_ toggle width = shared height, ≥44px on touch.
* **No visible seam / halves look like one button.** Users don't discover the menu exists, or they can't tell where "Save" ends and "▾" begins, so they never click the caret. _Fix:_ a ≥3:1 divider plus per-half hover/focus so two targets are obvious.
* **Wrong default in the primary slot.** If the big button does the _less_ common action, every user pays a menu-open tax for the thing they actually want, and some fire the wrong action by habit. _Fix:_ the primary must be the genuinely dominant action; validate with analytics, not assumption.
* **Destructive action as the default.** Putting "Delete" (vs "Archive") on the one-click half turns a stray click into data loss. _Fix:_ safe action is the default; destructive variants live in the menu with confirmation.
* **Menu doesn't close / doesn't return focus.** Open menu that survives outside-click, or Escape that dumps focus to the page top, strands keyboard and SR users and feels broken. _Fix:_ light-dismiss + Escape both close and return focus to the toggle.
* **Menu overflow-clipped inside a card/table.** The menu renders but gets cut off by an ancestor `overflow:hidden`, so options are unreadable. _Fix:_ render in the top layer (popover/portal) or with anchor positioning.
* **Loading state locks the wrong half.** Disabling the whole control during an async primary action prevents the user from opening "Cancel/Change" options that were the point. _Fix:_ decide per flow which half stays live; keep safe menu actions available where sensible.

## 13\. Advanced Patterns

* **TypeScript-enforced a11y via discriminated unions.** Make it impossible to ship an unlabeled toggle or an empty menu:

```plain
type SplitItem = { id: string; label: string; onSelect: () => void };
type SplitButtonProps =
  | { menuLabel: string; items: [SplitItem, ...SplitItem[]];   // non-empty tuple: menu must have ≥1 item
      primaryLabel: string; onPrimary: () => void; collapseToMenu?: false }
  | { menuLabel: string; items: [SplitItem, ...SplitItem[]];
      primaryLabel: string; onPrimary: () => void; collapseToMenu: true; primaryInMenu: true };
// The compiler rejects <SplitButton items={[]}> and a toggle with no menuLabel.
```

* **Design-token tiering (CSS custom properties).** Three-tier tokens keep the split button themeable and cross-platform: primitive → semantic → component.

```css
:root {
  --brand-700: #5F2C82;               /* primitive */
  --color-action: var(--brand-700);   /* semantic */
  --split-bg: var(--color-action);    /* component */
```

* **Controlled/uncontrolled open state.** Expose `open?`/`defaultOpen?` + `onOpenChange` so the split button works both self-managed and driven by a parent (e.g. a tutorial that forces the menu open), mirroring Radix conventions.
* **Single menu, two renderings (responsive collapse).** Drive both the desktop split and the mobile menu-button from one `items` array; on `< 360px` render a single menu button with the primary injected as the first item — no duplicated action logic.
* **`aria-live`** **result region.** After the primary/menu action resolves, write success/failure to a visually-associated `role="status"` region so SR users get async feedback without a focus jump.

## 14\. Performance & Bundle Cost

* **Compositor-only animations.** Animate only `transform` (caret rotate, active translate) and `opacity` (menu fade) — never `height`/`top`/`box-shadow` on open, which trigger layout/paint and jank the menu appearance. Add `will-change: transform` only while animating.
* **Lazy-mount the menu.** Render the `role="menu"` and its items only when open (conditional mount), not hidden in the DOM on every split button. On a toolbar or a data grid with 200 rows each carrying a split button, eager menus multiply DOM nodes and event listeners for menus almost never opened.
* **Tree-shakeable imports.** Import icons individually (`import { ChevronDown } from "lucide-react"`), not the whole icon set; import Radix/Headless menu primitives à la carte. A split button should add a few KB, not pull a full menu framework into a leaf component.
* **Drop Popper for native positioning.** CSS Anchor Positioning removes the runtime positioning library (Popper/Floating UI can be ~10KB) for browsers that support it — huge win when split buttons repeat across a table.
* **Delegate/pool listeners in lists.** For many split buttons, avoid per-instance `document` mousedown listeners; use one delegated outside-click handler or an `AbortController`\-scoped listener created only while a menu is open.
* **Debounce telemetry.** If you instrument primary vs menu usage, batch the events; don't fire a network call on every open.

## 15\. Security

Standalone. A split button is low client-side surface but is a genuine action dispatcher, so the server side matters.

* **Attack surface (client).** The control itself renders trusted, developer-authored labels — no user-generated content in the primary label or toggle by default, so no direct XSS vector in the chrome. IF menu item labels come from user/tenant data (e.g. "Save to {folderName}"), they must be escaped/`textContent`\-set, never `innerHTML`, to prevent stored XSS. The `items` JSON on the web-component example must be trusted or sanitized before parsing.
* **The real risk is the endpoint behind each action.** A split button multiplies the actions one control can trigger (`mode=default`, `save_as`, `save_as_template`…). The `mode`/action value is client-supplied and must be validated against an allow-list server-side (see §9.13) — never `eval`/dispatch on an arbitrary submitted string.
* **Authorization on privileged menu actions (the crux for gated menus).** Menu items frequently include elevated variants ("Save as template," "Deploy to production," "Delete permanently"). Hiding or disabling that item in the UI is NOT security — the endpoint must independently check the user's permission for that specific mode. A user can replay the POST with `mode=deploy_prod` regardless of what the menu showed. Authz check per action, server-side, every time.
* **CSRF + method.** Actions that mutate state must be POST (or use a mutation with a CSRF token / same-site cookies), not GET — a split button's default action is exactly the kind of one-click mutation an attacker would try to forge.
* **Confirmation, not just gating, for destructive variants.** Destructive menu items should require explicit confirmation and, for high-stakes ops, re-auth — the low-friction nature of the control is precisely why an accidental or forged destructive fire is dangerous.

## 16\. Senior-Level Checklist

Ship-ready gate — every item non-negotiable:

* **Semantics:** two real `<button>`s (primary + toggle) inside a `role="group"`; menu is `role="menu"` with `role="menuitem"` children — never one button, never a `<div onclick>`.
* **Toggle ARIA:** `aria-haspopup="menu"`, `aria-expanded` correctly toggled, `aria-controls` → menu id, and a descriptive accessible name (not "▾").
* **Keyboard:** Enter/Space fires primary; toggle opens menu (ArrowDown→first, ArrowUp→last); arrows cycle items with wrap; Home/End; typeahead; Escape closes.
* **Focus management:** focus moves into the menu on open, returns to the toggle on Escape/selection, and leaves cleanly on Tab/outside-click — never dropped to `<body>`.
* **Roving tabindex:** exactly one menuitem `tabindex="0"`, rest `-1`; menu container out of tab order.
* **Contrast:** labels ≥ 4.5:1; seam, caret, and focus ring ≥ 3:1; two halves distinguishable beyond the seam alone.
* **Target size:** each half ≥ 24px (AA) / ≥ 44px touch; toggle target = shared height; no overlapping hit areas.
* **States:** default/hover/focus/active/disabled/loading all present and per-half; disabled-primary-with-live-toggle (and vice versa) handled deliberately.
* **Reduced motion:** caret rotate + press translate + menu animation all respect `prefers-reduced-motion`.
* **Server side:** action `mode` validated against an allow-list; privileged menu actions authz-checked server-side; destructive variants confirmed; mutations are POST with CSRF protection.
* **Correct component:** verified it's genuinely primary-action-plus-variants (not a button group, segmented control, or menu button); menu has ≥2 items; "primary in menu?" decision made and consistent.
* **Responsive:** defined mobile behavior (collapse to menu button below the breakpoint) rather than shipping an unhittable caret.

## 17\. Visual Styles

The same behavioral skeleton, eleven skins. Each note is specific to how a split button renders in that language.

* **Flat** — solid single-color lozenge; the seam is a slightly lighter/darker 1px line, the caret a flat chevron, no shadows. The two halves read as one flat pill split by a hairline.
* **Material** — filled tonal button pair with a state-layer overlay per half and a ripple that originates within whichever half is pressed (never crossing the seam); caret is a 24dp `arrow_drop_down`; menu is an elevated surface (dp2) with its own ripple items.
* **Glassmorphism** — a frosted translucent lozenge over a blurred backdrop; the seam is a brighter specular hairline, the menu a second frosted panel with a subtle border so items stay legible through the blur.
* **Liquid Glass (2026, iOS 26 / Tahoe)** — the whole control is one continuous glass capsule; the seam becomes a refracted light-bend rather than a drawn line, and the caret half picks up a live specular highlight that intensifies on press. The design job is keeping the two targets legibly distinct under the refraction.
* **Neumorphism** — a soft extruded capsule on a same-color surface; the primary is a raised pad, the toggle a slightly recessed pad so the seam reads as a fold, and the open state presses the toggle inward (inset shadow). Contrast is the risk — pair with a real focus ring.
* **Skeuomorphism** — a beveled physical button with a gradient sheen and a routed groove for the seam, like a two-part hardware key; the caret is an embossed triangle and the menu drops like a physical tray with a drop shadow.
* **Neo-Brutalism** — thick black border around both halves, a hard 2–3px offset drop shadow, no radius or a tiny one; the seam is a full-weight black rule and the caret a chunky glyph; menu is a bordered box with the same brutal offset shadow.
* **Claymorphism** — puffy rounded 3D "clay" lozenge with soft inner+outer shadows; the two halves look like two pressable clay lobes joined at a soft valley (the seam), the caret debossed into the clay.
* **Aurora / Gradient** — animated multi-stop gradient fill (Eminence→Spanish Orange) across the whole lozenge; the seam is a thin darkened band so the gradient reads continuous, and the focus ring is a solid color so it survives against the moving background.
* **Minimal / Swiss** — near-invisible chrome: text primary + a hairline seam + a small caret, generous whitespace, one accent color only on hover/focus. Typography and alignment carry it; the menu is a clean bordered list, no icons.
* **UJG Brand (house default)** — Goldenrod (`#DCA424`) caret and label on an Eminence (`#5F2C82`) fill over Night (`#0A0A0A`); the seam is a 1px Goldenrod-at-28% hairline, the focus ring solid Goldenrod at 2px offset, the menu a Night panel with Eminence hover on items. Mallong flourish reserved for Amazonia-persona surfaces only.

**The rule that never changes:** style is skin, behavior is the skeleton. Across all eleven skins the semantics (two buttons, `role="menu"`), the keyboard model (Enter/Space fires primary, toggle opens menu, arrows cycle, Escape closes and returns focus), the focus ring, and the ≥44px targets are identical. Change how it looks; never change how it behaves.
